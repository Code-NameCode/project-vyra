    68→/* ---- minimal Web Speech API typing (no `any`) ---- */
    69→interface SpeechResultLike {
    70→  0: { transcript: string };
    71→  isFinal: boolean;
    72→  length: number;
    73→}
    74→interface SpeechEventLike {
    75→  results: ArrayLike<SpeechResultLike>;
    76→}
    77→interface SpeechRecognitionLike {
    78→  lang: string;
    79→  continuous: boolean;
    80→  interimResults: boolean;
    81→  start: () => void;
    82→  stop: () => void;
    83→  abort: () => void;
    84→  onresult: ((event: SpeechEventLike) => void) | null;
    85→  onerror: ((event: unknown) => void) | null;
    86→  onend: (() => void) | null;
    87→}
    88→type SpeechRecognitionCtor = new () => SpeechRecognitionLike;
    89→type WindowWithSpeech = Window & {
    90→  SpeechRecognition?: SpeechRecognitionCtor;
    91→  webkitSpeechRecognition?: SpeechRecognitionCtor;
    92→};

    94→/* ---- Plan / Build mode ---- */
    95→type BuildMode = "plan" | "build";
    96→// Other modules may extend Conversation/ChatState with buildMode. Access
    97→// defensively so the Composer keeps working whether or not the store has
    98→// been augmented yet.
    99→type ConvWithBuild = Conversation & { buildMode?: BuildMode };
   100→interface ChatStoreWithBuild {
   101→  setBuildMode?: (tabId: string, mode: BuildMode) => void;
   102→}

   140→/* ---- @ context reference item ---- */
   141→type SlashCommandWithIcon = SlashCommand & { icon?: string };
   142→interface RefItem {
   143→  kind: "file" | "project";
   144→  name: string;
   145→  path: string;
   146→  icon: LucideIcon;
   147→}
   148→const EMPTY_FILES: FileNode[] = [];
   149→const EMPTY_SKILLS: string[] = [];
   150→
   151→/** Strip a leading `/` from a slash command label (e.g. "/review" -> "review"). */
   152→function stripSlash(label: string): string {
   153→  return label.replace(/^\//, "");
   154→}

   156→/**
   157→ * Composer — the core prompt input.
   158→ * Auto-growing textarea, slash command popup (with skill icons) that adds
   159→ * action skills as removable chips, Codex-style `@` context referencing,
   160→ * attach menu, Plan/Build mode selector, agent picker chip, model picker
   161→ * chip, and a send/stop/voice button.
   162→ */
   163→export function Composer({ tabId }: { tabId: string }) {
   164→  const ensure = useChatStore((s) => s.ensure);
   165→  const draft = useChatStore((s) => s.conversations[tabId]?.composerDraft ?? "");
   166→  const streaming = useChatStore(
   167→    (s) => s.conversations[tabId]?.streaming ?? false
   168→  );
   169→  const convModelId = useChatStore((s) => s.conversations[tabId]?.modelId);
   170→  const defaultModelId = useSettingsStore((s) => s.defaultModelId);
   171→  const setComposerDraft = useChatStore((s) => s.setComposerDraft);
   172→  const sendMessage = useChatStore((s) => s.sendMessage);
   173→  const stopStreaming = useChatStore((s) => s.stopStreaming);
   174→  const clearConversation = useChatStore((s) => s.clearConversation);
   175→
   176→  /* ---- skills (multi-chip) ---- */
   177→  const skills = useChatStore(
   178→    (s) => s.conversations[tabId]?.skills ?? EMPTY_SKILLS
   179→  );
   180→  const addSkill = useChatStore((s) => s.addSkill);
   181→  const removeSkill = useChatStore((s) => s.removeSkill);
   182→
   183→  /* ---- agent picker ---- */
   184→  const agentId = useChatStore(
   185→    (s) => s.conversations[tabId]?.agentId ?? ""
   186→  );
   187→  const setAgent = useChatStore((s) => s.setAgent);
   188→
   189→  /* ---- Deep research mode ---- */
   190→  const researchMode = useChatStore(
   191→    (s) => s.conversations[tabId]?.researchMode ?? false
   192→  );
   193→  const setResearchMode = useChatStore((s) => s.setResearchMode);
   194→
   195→  /* ---- Plan / Build mode (defensive: works with or without store support) ---- */
   196→  const storeBuildMode = useChatStore(
   197→    (s) =>
   198→      (s.conversations[tabId] as ConvWithBuild | undefined)?.buildMode ??
   199→      ("build" as BuildMode)
   200→  );
   201→  const storeSetBuildMode = useChatStore(
   202→    (s) => (s as unknown as ChatStoreWithBuild).setBuildMode
   203→  );
   204→  const [localBuildMode, setLocalBuildMode] =
   205→    React.useState<BuildMode>("build");
   206→  const buildMode: BuildMode = storeSetBuildMode
   207→    ? (storeBuildMode as BuildMode)
   208→    : localBuildMode;
   209→  const setBuildMode = React.useCallback(
   210→    (mode: BuildMode) => {
   211→      if (storeSetBuildMode) storeSetBuildMode(tabId, mode);
   212→      else setLocalBuildMode(mode);
   213→    },
   214→    [storeSetBuildMode, tabId]
   215→  );
   216→
   217→  /* ---- active workspace (reactive) for `@` referencing ---- */
   218→  const projectId = useAppStore(
   219→    (s) => s.tabs.find((t) => t.id === tabId)?.projectId
   220→  );
   221→  const projects = useProjectsStore((s) => s.projects);
   222→  const setTabProject = useAppStore((s) => s.setTabProject);
   223→  const addProject = useProjectsStore((s) => s.addProject);
   224→  const activeProject = projects.find((p) => p.id === projectId);
   225→  const convFiles = useChatStore(
   226→    (s) => s.conversations[tabId]?.workspace.files
   227→  );
   228→
   229→  const files = React.useMemo<FileNode[]>(() => {
   230→    if (projectId) {
   231→      return (
   232→        projects.find((p) => p.id === projectId)?.workspace.files ??
   233→        EMPTY_FILES
   234→      );
   235→    }
   236→    return convFiles ?? EMPTY_FILES;
   237→  }, [projectId, projects, convFiles]);
   238→
   239→  const refItems = React.useMemo<RefItem[]>(() => {
   240→    const fileItems: RefItem[] = files.map((f) => ({
   241→      kind: "file",
   242→      name: f.name,
   243→      path: f.path,
   244→      icon: File,
   245→    }));
   246→    const projItems: RefItem[] = projects.map((p) => ({
   247→      kind: "project",
   248→      name: p.name,
   249→      path: p.name,
   250→      icon: Folder,
   251→    }));
   252→    return [...fileItems, ...projItems];
   253→  }, [files, projects]);
   254→
   255→  const modelId = convModelId ?? defaultModelId;
   256→
   257→  const taRef = React.useRef<HTMLTextAreaElement>(null);
   258→  const odelOpen, setModelOpen] = React.useState(false);
   259→  const [agentOpen, setAgentOpen] = React.useState(false);
   260→  const [slashSuppressed, setSlashSuppressed] = React.useState(false);
   261→  const [selIndex, setSelIndex] = React.useState(0);
   262→  const [atSuppressed, setAtSuppressed] = React.useState(false);
   263→  const [atIndex, setAtIndex] = React.useState(0);
   264→
   265→  // ensure a conversation object exists so modelId/draft are stable
   266→  React.useEffect(() => {
   267→    ensure(tabId);
   268→  }, [tabId, ensure]);
   269→
   270→  // auto-grow textarea whenever draft changes (covers typing + external updates)
   271→  React.useLayoutEffect(() => {
   272→    const ta = taRef.current;
   273→    if (!ta) return;
   274→    ta.style.height = "auto";
   275→    ta.style.height = `${Math.min(ta.scrollHeight, 200)}px`;
   276→  }, [draft]);
   277→
   278→  // reset slash + @ suppression + selection whenever draft changes
   279→  React.useEffect(() => {
   280→    setSlashSuppressed(false);
   281→    setSelIndex(0);
   282→    setAtSuppressed(false);
   283→    setAtIndex(0);
   284→  }, [draft]);
   285→
   286→  /* ---- slash command popup derivation ---- */
   287→  const slashWithIcons = React.useMemo<SlashCommandWithIcon[]>(
   288→    () => SLASH_COMMANDS as SlashCommandWithIcon[],
   289→    []
   290→  );
   291→  const isSlashDraft =
   292→    draft.startsWith("/") &&
   293→    !draft.slice(1).includes(" ") &&
   294→    !streaming;
   295→  const filtered = React.useMemo<SlashCommand[]>(() => {
   296→    if (!isSlashDraft) return [];
   297→    const q = draft.slice(1).toLowerCase();
   298→    if (!q) return slashWithIcons;
   299→    return slashWithIcons.filter(
   300→      (c) =>
   301→        c.label.toLowerCase().includes(q) ||
   302→        c.id.toLowerCase().includes(q) ||
   303→        c.hint.toLowerCase().includes(q)
   304→    );
   305→  }, [isSlashDraft, draft, slashWithIcons]);
   306→  const slashVisible = isSlashDraft && !slashSuppressed && filtered.length > 0;
   307→
   308→  /* ---- @ context reference popup derivation ---- */
   309→  const atQuery = React.useMemo<{ index: number; query: string } | null>(() => {
   310→    const lastAt = draft.lastIndexOf("@");
   311→    if (lastAt === -1) return null;
   312→    const after = draft.slice(lastAt + 1);
   313→    // the query is the text from the last "@" to the end of the draft;
   314→    // any whitespace terminates it.
   315→    if (/\s/.test(after)) return null;
   316→    return { index: lastAt, query: after };
   317→  }, [draft]);
   318→
   319→  const atFiltered = React.useMemo<RefItem[]>(() => {
   320→    if (!atQuery) return [];
   321→    const q = atQuery.query.toLowerCase();
   322→    const list = q
   323→      ? refItems.filter(
   324→          (it) =>
   325→            it.name.toLowerCase().includes(q) ||
   326→            it.path.toLowerCase().includes(q)
   327→        )
   328→      : refItems;
   329→    return list.slice(0, 12);
   330→  }, [atQuery, refItems]);
   331→
   332→  const atVisible =
   333→    !streaming &&
   334→    !slashVisible &&
   335→    atQuery !== null &&
   336→    !atSuppressed &&
   337→    atFiltered.length > 0;
   338→
   339→  // keep atIndex within bounds when the filtered list shrinks
   340→  React.useEffect(() => {
   341→    if (atFiltered.length > 0 && atIndex >= atFiltered.length) {
   342→      setAtIndex(0);
   343→    }
   344→  }, [atFiltered, atIndex]);
   345→
   346→  /* ---- actions ---- */
   347→  const handleSend = React.useCallback(() => {
   348→    if (streaming) return;
   349→    const text = useChatStore.getState().conversations[tabId]?.composerDraft ?? "";
   350→    if (!text.trim()) return;
   351→    void sendMessage(tabId, text, modelId);
   352→  }, [streaming, tabId, modelId, sendMessage]);
   353→
   354→  const selectCommand = React.useCallback(
   355→    (cmd: SlashCommand) => {
   356→      // Read the live draft + skills from the store to avoid stale closures.
   357→      const currentDraft =
   358→        useChatStore.getState().conversations[tabId]?.composerDraft ?? "";
   359→      const beforeSlash = currentDraft.slice(0, currentDraft.lastIndexOf("/"));
   360→      const focusTextarea = () =>
   361→        requestAnimationFrame(() => taRef.current?.focus());
   362→
   363→      if (cmd.id === "model") {
   364→        setComposerDraft(tabId, "");
   365→        setModelOpen(true);
   366→        return;
   367→      }
   368→      if (cmd.id === "clear") {
   369→        clearConversation(tabId);
   370→        setComposerDraft(tabId, "");
   371→        toast("Conversation cleared");
   372→        return;
   373→      }
   374→      if (cmd.id === "help") {
   375→        setComposerDraft(tabId, "");
   376→        toast("Press / for commands · Enter to send · Shift+Enter for newline", {
   377→          duration: 4000,
   378→        });
   379→        return;
   380→      }
   381→      if (cmd.id === "plan") {
   382→        setBuildMode("plan");
   383→        setComposerDraft(tabId, beforeSlash);
   384→        toast("Switched to Plan");
   385→        focusTextarea();
   386→        return;
   387→      }
   388→      if (cmd.id === "build") {
   389→        setBuildMode("build");
   390→        setComposerDraft(tabId, beforeSlash);
   391→        toast("Switched to Build");
   392→        focusTextarea();
   393→        return;
   394→      }
   395→      // action skills (review, test, fix, explain, refactor, document, init):
   396→      // add as a removable chip instead of inserting text.
   397→      const currentSkills =
   398→        useChatStore.getState().conversations[tabId]?.skills ?? [];
   399→      if (currentSkills.includes(cmd.id)) {
   400→        toast("Already added");
   401→      } else {
   402→        addSkill(tabId, cmd.id);
   403→        toast(`${stripSlash(cmd.label)} skill added`);
   404→      }
   405→      setComposerDraft(tabId, beforeSlash);
   406→      focusTextarea();
   407→    },
   408→    [
   409→      tabId,
   410→      setComposerDraft,
   411→      clearConversation,
   412→      addSkill,
   413→      setBuildMode,
   414→    ]
   415→  );
   416→
   417→  const selectRef = React.useCallback(
   418→    (item: RefItem) => {
   419→      if (!atQuery) return;
   420→      const before = draft.slice(0, atQuery.index);
   421→      const next = `${before}@${item.name} `;
   422→      setComposerDraft(tabId, next);
   423→      requestAnimationFrame(() => taRef.current?.focus());
   424→    },
   425→    [atQuery, draft, setComposerDraft, tabId]
   426→  );
   427→
   428→  /* ---- keyboard ---- */
   429→  const onKeyDown = (e: React.KeyboardEvent<HTMLTextAreaElement>) => {
   430→    if (slashVisible) {
   431→      if (e.key === "ArrowDown") {
   432→        e.preventDefault();
   433→        setSelIndex((i) => (i + 1) % filtered.length);
   434→        return;
   435→      }
   436→      if (e.key === "ArrowUp") {
   437→        e.preventDefault();
   438→        setSelIndex((i) => (i - 1 + filtered.length) % filtered.length);
   439→        return;
   440→      }
   441→      if (e.key === "Enter") {
   442→        e.preventDefault();
   443→        const cmd = filtered[selIndex] ?? filtered[0];
   444→        if (cmd) selectCommand(cmd);
   445→        return;
   446→      }
   447→      if (e.key === "Escape") {
   448→        e.preventDefault();
   449→        setSlashSuppressed(true);
   450→        return;
   451→      }
   452→      if (e.key === "Tab") {
   453→        e.preventDefault();
   454→        const cmd = filtered[selIndex] ?? filtered[0];
   455→        if (cmd) selectCommand(cmd);
   456→        return;
   457→      }
   458→    }
   459→    if (atVisible) {
   460→      if (e.key === "ArrowDown") {
   461→        e.preventDefault();
   462→        setAtIndex((i) => (i + 1) % atFiltered.length);
   463→        return;
   464→      }
   465→      if (e.key === "ArrowUp") {
   466→        e.preventDefault();
   467→        setAtIndex((i) => (i - 1 + atFiltered.length) % atFiltered.length);
   468→        return;
   469→      }
   470→      if (e.key === "Enter" || e.key === "Tab") {
   471→        e.preventDefault();
   472→        const item = atFiltered[atIndex] ?? atFiltered[0];
   473→        if (item) selectRef(item);
   474→        return;
   475→      }
   476→      if (e.key === "Escape") {
   477→        e.preventDefault();
   478→        setAtSuppressed(true);
   479→        return;
   480→      }
   481→    }
   482→    if (e.key === "Enter" && !e.shiftKey && !e.nativeEvent.isComposing) {
   483→      if (!streaming && draft.trim()) {
   484→        e.preventDefault();
   485→        handleSend();
   486→      }
   487→    }
   488→  };
   489→
   490→  /* ---- voice ---- */
   491→  const startVoice = React.useCallback(() => {
   492→    const w = window as WindowWithSpeech;
   493→    const Ctor = w.SpeechRecognition ?? w.webkitSpeechRecognition;
   494→    if (!Ctor) {
   495→      toast.error("Voice input not supported in this browser");
   496→      return;
   497→    }
   498→    try {
   499→      const rec = new Ctor();
   500→      rec.lang = "en-US";
   501→      rec.continuous = false;
   502→      rec.interimResults = false;
   503→      rec.onresult = (event: SpeechEventLike) => {
   504→        let transcript = "";
   505→        for (let i = 0; i < event.results.length; i++) {
   506→          transcript += event.results[i][0].transcript;
   507→        }
   508→        if (!transcript) return;
   509→        const cur =
   510→          useChatStore.getState().conversations[tabId]?.composerDraft ?? "";
   511→        const next = (cur ? cur + " " : "") + transcript.trim();
   512→        setComposerDraft(tabId, next);
   513→      };
   514→      rec.onerror = () => {
   515→        toast.error("Voice input failed");
   516→      };
   517→      rec.start();
   518→      toast("Listening…", { duration: 2500 });
   519→    } catch {
   520→      toast.error("Voice input could not start");
   521→    }
   522→  }, [tabId, setComposerDraft]);
   523→
   524→  /* ---- send button state ---- */
   525→  const canSend = draft.trim().length > 0;
   526→  const sendState: "send" | "stop" | "voice" = streaming
   527→    ? "stop"
   528→    : canSend
   529→    ? "send"
   530→    : "voice";
   531→
   532→  const onSendClick = () => {
   533→    if (sendState === "stop") stopStreaming(tabId);
   534→    else if (sendState === "send") handleSend();
   535→    else startVoice();
   536→  };
   537→
   538→  const SendIcon = sendState === "stop" ? Square : sendState === "send" ? ArrowUp : Mic;
   539→  const sendLabel = sendState === "stop" ? "Stop" : sendState === "send" ? "Send" : "Voice";
   540→
   541→  /* ---- active agent (for the trigger chip) ---- */
   542→  const activeAgent: AgentDef | undefined = agentId
   543→    ? AGENTS.find((a) => a.id === agentId)
   544→    : undefined;
   545→  const agentTriggerName = activeAgent ? activeAgent.name : "Agent";
   546→
   547→  const handleAgentSelect = (id: string, name: string) => {
   548→    setAgent(tabId, id);
   549→    setAgentOpen(false);
   550→    toast(`${name} selected`);
   551→  };
   552→
   553→  return (
   554→    <div className="relative w-full">
   555→      {/* slash command popup (above the box) */}
   556→      {slashVisible && (
   557→        <div
   558→          className={cn(
   559→            "absolute bottom-full mb-2 left-0 w-[280px] max-w-[calc(100vw-2rem)] z-30",
   560→            "rounded-[12px] border border-ws-border bg-popover p-1.5 ws-fade-in",
   561→            "shadow-2xl max-h-[260px] overflow-y-auto ws-scroll"
   562→          )}
   563→          role="listbox"
   564→        >
   565→          {filtered.map((c, i) => {
   566→            const Icon = skillIcon(
   567→              (c as SlashCommandWithIcon).icon
   568→            );
   569→            return (
   570→              <button
   571→                type="button"
   572→                key={c.id}
   573→                onMouseEnter={() => setSelIndex(i)}
   574→                onClick={() => selectCommand(c)}
   575→                className={cn(
   576→                  "w-full flex items-center gap-2.5 rounded-[8px] px-2.5 py-2 cursor-pointer",
   577→                  "transition-colors text-left",
   578→                  i === selIndex ? "bg-hover" : "hover:bg-hover"
   579→                )}
   580→              >
   581→                <Icon
   582→                  className={cn(
   583→                    "size-4 shrink-0",
   584→                    i === selIndex ? "text-accent-hi" : "text-ws-secondary"
   585→                  )}
   586→                />
   587→                <div className="min-w-0">
   588→                  <div className="text-[13px] font-medium text-foreground">
   589→                    {c.label}
   590→                  </div>
   591→                  <div className="text-[12px] text-ws-secondary line-clamp-1">
   592→                    {c.description}
   593→                  </div>
   594→                </div>
   595→              </button>
