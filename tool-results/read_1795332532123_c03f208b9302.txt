    65→    65→/* ---- minimal Web Speech API typing (no `any`) ---- */
    66→    66→interface SpeechResultLike {
    67→    67→  0: { transcript: string };
    68→    68→  isFinal: boolean;
    69→    69→  length: number;
    70→    70→}
    71→    71→interface SpeechEventLike {
    72→    72→  results: ArrayLike<SpeechResultLike>;
    73→    73→}
    74→    74→interface SpeechRecognitionLike {
    75→    75→  lang: string;
    76→    76→  continuous: boolean;
    77→    77→  interimResults: boolean;
    78→    78→  start: () => void;
    79→    79→  stop: () => void;
    80→    80→  abort: () => void;
    81→    81→  onresult: ((event: SpeechEventLike) => void) | null;
    82→    82→  onerror: ((event: unknown) => void) | null;
    83→    83→  onend: (() => void) | null;
    84→    84→}
    85→    85→type SpeechRecognitionCtor = new () => SpeechRecognitionLike;
    86→    86→type WindowWithSpeech = Window & {
    87→    87→  SpeechRecognition?: SpeechRecognitionCtor;
    88→    88→  webkitSpeechRecognition?: SpeechRecognitionCtor;
    89→    89→};

   145→   145→const EMPTY_FILES: FileNode[] = [];

   153→   153→/**
   154→   154→ * Composer — the core prompt input.
   155→   155→ * Auto-growing textarea, slash command popup (with skill icons) that adds
   156→   156→ * action skills as removable chips, Codex-style `@` context referencing,
   157→   157→ * attach menu, Plan/Build mode selector, agent picker chip, model picker
   158→   158→ * chip, and a send/stop/voice button.
   159→   159→ */
   160→   160→export function Composer({ tabId }: { tabId: string }) {
   161→   161→  const ensure = useChatStore((s) => s.ensure);
   162→   162→  const draft = useChatStore((s) => s.conversations[tabId]?.composerDraft ?? "");
   163→   163→  const streaming = useChatStore(
   164→   164→    (s) => s.conversations[tabId]?.streaming ?? false
   165→   165→  );
   166→   166→  const convModelId = useChatStore((s) => s.conversations[tabId]?.modelId);
   167→   167→  const defaultModelId = useSettingsStore((s) => s.defaultModelId);
   168→   168→  const setComposerDraft = useChatStore((s) => s.setComposerDraft);
   169→   169→  const sendMessage = useChatStore((s) => s.sendMessage);
   170→   170→  const stopStreaming = useChatStore((s) => s.stopStreaming);
   171→   171→  const clearConversation = useChatStore((s) => s.clearConversation);
   172→   172→
   173→   173→  /* ---- skills (multi-chip) ---- */
   174→   174→  const skills = useChatStore(
   175→   175→    (s) => s.conversations[tabId]?.skills ?? EMPTY_SKILLS
   176→   176→  );
   177→   177→  const addSkill = useChatStore((s) => s.addSkill);
   178→   178→  const removeSkill = useChatStore((s) => s.removeSkill);
   179→   179→
   180→   180→  /* ---- agent picker ---- */
   181→   181→  const agentId = useChatStore(
   182→   182→    (s) => s.conversations[tabId]?.agentId ?? ""
   183→   183→  );
   184→   184→  const setAgent = useChatStore((s) => s.setAgent);
   185→   185→
   186→   186→  /* ---- Plan / Build mode (defensive: works with or without store support) ---- */
   187→   187→  const storeBuildMode = useChatStore(
   188→   188→    (s) =>
   189→   189→      (s.conversations[tabId] as ConvWithBuild | undefined)?.buildMode ??
   190→   190→      ("build" as BuildMode)
   191→   191→  );
   192→   192→  const storeSetBuildMode = useChatStore(
   193→   193→    (s) => (s as unknown as ChatStoreWithBuild).setBuildMode
   194→   194→  );
   195→   195→  const [localBuildMode, setLocalBuildMode] =
   196→   196→    React.useState<BuildMode>("build");
   197→   197→  const buildMode: BuildMode = storeSetBuildMode
   198→   198→    ? (storeBuildMode as BuildMode)
   199→   199→    : localBuildMode;
   200→   200→  const setBuildMode = React.useCallback(
   201→   201→    (mode: BuildMode) => {
   202→   202→      if (storeSetBuildMode) storeSetBuildMode(tabId, mode);
   203→   203→      else setLocalBuildMode(mode);
   204→   204→    },
   205→   205→    [storeSetBuildMode, tabId]
   206→   206→  );
   207→   207→
   208→   208→  /* ---- active workspace (reactive) for `@` referencing ---- */
   209→   209→  const projectId = useAppStore(
   210→   210→    (s) => s.tabs.find((t) => t.id === tabId)?.projectId
   211→   211→  );
   212→   212→  const projects = useProjectsStore((s) => s.projects);
   213→   213→  const convFiles = useChatStore(
   214→   214→    (s) => s.conversations[tabId]?.workspace.files
   215→   215→  );
   216→   216→
   217→   217→  const files = React.useMemo<FileNode[]>(() => {
   218→   218→    if (projectId) {
   219→   219→      return (
   220→   220→        projects.find((p) => p.id === projectId)?.workspace.files ??
   221→   221→        EMPTY_FILES
   222→   222→      );
   223→   223→    }
   224→   224→    return convFiles ?? EMPTY_FILES;
   225→   225→  }, [projectId, projects, convFiles]);
   226→   226→
   227→   227→  const refItems = React.useMemo<RefItem[]>(() => {
   228→   228→    const fileItems: RefItem[] = files.map((f) => ({
   229→   229→      kind: "file",
   230→   230→      name: f.name,
   231→   231→      path: f.path,
   232→   232→      icon: File,
   233→   233→    }));
   234→   234→    const projItems: RefItem[] = projects.map((p) => ({
   235→   235→      kind: "project",
   236→   236→      name: p.name,
   237→   237→      path: p.name,
   238→   238→      icon: Folder,
   239→   239→    }));
   240→   240→    return [...fileItems, ...projItems];
   241→   241→  }, [files, projects]);
   242→   242→
   243→   243→  const modelId = convModelId ?? defaultModelId;
   244→   244→
   245→   245→  const taRef = React.useRef<HTMLTextAreaElement>(null);
   246→   246→  const [modelOpen, setModelOpen] = React.useState(false);
   247→   247→  const [agentOpen, setAgentOpen] = React.useState(false);
   248→   248→  const [slashSuppressed, setSlashSuppressed] = React.useState(false);
   249→   249→  const [selIndex, setSelIndex] = React.useState(0);
   250→   250→  const [atSuppressed, setAtSuppressed] = React.useState(false);
   251→   251→  const [atIndex, setAtIndex] = React.useState(0);
   252→   252→
   253→   253→  // ensure a conversation object exists so modelId/draft are stable
   254→   254→  React.useEffect(() => {
   255→   255→    ensure(tabId);
   256→   256→  }, [tabId, ensure]);
   257→   257→
   258→   258→  // auto-grow textarea whenever draft changes (covers typing + external updates)
   259→   259→  React.useLayoutEffect(() => {
   260→   260→    const ta = taRef.current;
   261→   261→    if (!ta) return;
   262→   262→    ta.style.height = "auto";
   263→   263→    ta.style.height = `${Math.min(ta.scrollHeight, 200)}px`;
   264→   264→  }, [draft]);
   265→   265→
   266→   266→  // reset slash + @ suppression + selection whenever draft changes
   267→   267→  React.useEffect(() => {
   268→   268→    setSlashSuppressed(false);
   269→   269→    setSelIndex(0);
   270→   270→    setAtSuppressed(false);
   271→   271→    setAtIndex(0);
   272→   272→  }, [draft]);
   273→   273→
   274→   274→  /* ---- slash command popup derivation ---- */
   275→   275→  const slashWithIcons = React.useMemo<SlashCommandWithIcon[]>(
   276→   276→    () => SLASH_COMMANDS as SlashCommandWithIcon[],
   277→   277→    []
   278→   278→  );
   279→   279→  const isSlashDraft =
   280→   280→    draft.startsWith("/") &&
   281→   281→    !draft.slice(1).includes(" ") &&
   282→   282→    !streaming;
   283→   283→  const filtered = React.useMemo<SlashCommand[]>(() => {
   284→   284→    if (!isSlashDraft) return [];
   285→   285→    const q = draft.slice(1).toLowerCase();
   286→   286→    if (!q) return slashWithIcons;
   287→   287→    return slashWithIcons.filter(
   288→   288→      (c) =>
   289→   289→        c.label.toLowerCase().includes(q) ||
   290→   290→        c.id.toLowerCase().includes(q) ||
   291→   291→        c.hint.toLowerCase().includes(q)
   292→   292→    );
   293→   293→  }, [isSlashDraft, draft, slashWithIcons]);
   294→   294→  const slashVisible = isSlashDraft && !slashSuppressed && filtered.length > 0;
   295→   295→
   296→   296→  /* ---- @ context reference popup derivation ---- */
   297→   297→  const atQuery = React.useMemo<{ index: number; query: string } | null>(() => {
   298→   298→    const lastAt = draft.lastIndexOf("@");
   299→   299→    if (lastAt === -1) return null;
   300→   300→    const after = draft.slice(lastAt + 1);
   301→   301→    // the query is the text from the last "@" to the end of the draft;
   302→   302→    // any whitespace terminates it.
   303→   303→    if (/\s/.test(after)) return null;
   304→   304→    return { index: lastAt, query: after };
   305→   305→  }, [draft]);
   306→   306→
   307→   307→  const atFiltered = React.useMemo<RefItem[]>(() => {
   308→   308→    if (!atQuery) return [];
   309→   309→    const q = atQuery.query.toLowerCase();
   310→   310→    const list = q
   311→   311→      ? refItems.filter(
   312→   312→          (it) =>
   313→   313→            it.name.toLowerCase().includes(q) ||
   314→   314→            it.path.toLowerCase().includes(q)
   315→   315→        )
   316→   316→      : refItems;
   317→   317→    return list.slice(0, 12);
   318→   318→  }, [atQuery, refItems]);
   319→   319→
   320→   320→  const atVisible =
   321→   321→    !streaming &&
   322→   322→    !slashVisible &&
   323→   323→    atQuery !== null &&
   324→   324→    !atSuppressed &&
   325→   325→    atFiltered.length > 0;
   326→   326→
   327→   327→  // keep atIndex within bounds when the filtered list shrinks
   328→   328→  React.useEffect(() => {
   329→   329→    if (atFiltered.length > 0 && atIndex >= atFiltered.length) {
   330→   330→      setAtIndex(0);
   331→   331→    }
   332→   332→  }, [atFiltered, atIndex]);
   333→   333→
   334→   334→  /* ---- actions ---- */
   335→   335→  const handleSend = React.useCallback(() => {
   336→   336→    if (streaming) return;
   337→   337→    const text = useChatStore.getState().conversations[tabId]?.composerDraft ?? "";
   338→   338→    if (!text.trim()) return;
   339→   339→    void sendMessage(tabId, text, modelId);
   340→   340→  }, [streaming, tabId, modelId, sendMessage]);
   341→   341→
   342→   342→  const selectCommand = React.useCallback(
   343→   343→    (cmd: SlashCommand) => {
   344→   344→      // Read the live draft + skills from the store to avoid stale closures.
   345→   345→      const currentDraft =
   346→   346→        useChatStore.getState().conversations[tabId]?.composerDraft ?? "";
   347→   347→      const beforeSlash = currentDraft.slice(0, currentDraft.lastIndexOf("/"));
   348→   348→      const focusTextarea = () =>
   349→   349→        requestAnimationFrame(() => taRef.current?.focus());
   350→   350→
   351→   351→      if (cmd.id === "model") {
   352→   352→        setComposerDraft(tabId, "");
   353→   353→        setModelOpen(true);
   354→   354→        return;
   355→   355→      }
   356→   356→      if (cmd.id === "clear") {
   357→   357→        clearConversation(tabId);
   358→   358→        setComposerDraft(tabId, "");
   359→   359→        toast("Conversation cleared");
   360→   360→        return;
   361→   361→      }
   362→   362→      if (cmd.id === "help") {
   363→   363→        setComposerDraft(tabId, "");
   364→   364→        toast("Press / for commands · Enter to send · Shift+Enter for newline", {
   365→   365→          duration: 4000,
   366→   366→        });
   367→   367→        return;
   368→   368→      }
   369→   369→      if (cmd.id === "plan") {
   370→   370→        setBuildMode("plan");
   371→   371→        setComposerDraft(tabId, beforeSlash);
   372→   372→        toast("Switched to Plan");
   373→   373→        focusTextarea();
   374→   374→        return;
   375→   375→      }
   376→   376→      if (cmd.id === "build") {
   377→   377→        setBuildMode("build");
   378→   378→        setComposerDraft(tabId, beforeSlash);
   379→   379→        toast("Switched to Build");
   380→   380→        focusTextarea();
   381→   381→        return;
   382→   382→      }
   383→   383→      // action skills (review, test, fix, explain, refactor, document, init):
   384→   384→      // add as a removable chip instead of inserting text.
   385→   385→      const currentSkills =
   386→   386→        useChatStore.getState().conversations[tabId]?.skills ?? [];
   387→   387→      if (currentSkills.includes(cmd.id)) {
   388→   388→        toast("Already added");
   389→   389→      } else {
   390→   390→        addSkill(tabId, cmd.id);
   391→   391→        toast(`${stripSlash(cmd.label)} skill added`);
   392→   392→      }
   393→   393→      setComposerDraft(tabId, beforeSlash);
   394→   394→      focusTextarea();
   395→   395→    },
   396→   396→    [
   397→   397→      tabId,
   398→   398→      setComposerDraft,
   399→   399→      clearConversation,
   400→   400→      addSkill,
   401→   401→      setBuildMode,
   402→   402→    ]
   403→   403→  );
   404→   404→
   405→   405→  const selectRef = React.useCallback(
   406→   406→    (item: RefItem) => {
   407→   407→      if (!atQuery) return;
   408→   408→      const before = draft.slice(0, atQuery.index);
   409→   409→      const next = `${before}@${item.name} `;
   410→   410→      setComposerDraft(tabId, next);
   411→   411→      requestAnimationFrame(() => taRef.current?.focus());
   412→   412→    },
   413→   413→    [atQuery, draft, setComposerDraft, tabId]
   414→   414→  );
   415→   415→
   416→   416→  /* ---- keyboard ---- */
   417→   417→  const onKeyDown = (e: React.KeyboardEvent<HTMLTextAreaElement>) => {
   418→   418→    if (slashVisible) {
   419→   419→      if (e.key === "ArrowDown") {
   420→   420→        e.preventDefault();
   421→   421→        setSelIndex((i) => (i + 1) % filtered.length);
   422→   422→        return;
   423→   423→      }
   424→   424→      if (e.key === "ArrowUp") {
   425→   425→        e.preventDefault();
   426→   426→        setSelIndex((i) => (i - 1 + filtered.length) % filtered.length);
   427→   427→        return;
   428→   428→      }
   429→   429→      if (e.key === "Enter") {
   430→   430→        e.preventDefault();
   431→   431→        const cmd = filtered[selIndex] ?? filtered[0];
   432→   432→        if (cmd) selectCommand(cmd);
   433→   433→        return;
   434→   434→      }
   435→   435→      if (e.key === "Escape") {
   436→   436→        e.preventDefault();
   437→   437→        setSlashSuppressed(true);
   438→   438→        return;
   439→   439→      }
   440→   440→      if (e.key === "Tab") {
   441→   441→        e.preventDefault();
   442→   442→        const cmd = filtered[selIndex] ?? filtered[0];
   443→   443→        if (cmd) selectCommand(cmd);
   444→   444→        return;
   445→   445→      }
   446→   446→    }
   447→   447→    if (atVisible) {
   448→   448→      if (e.key === "ArrowDown") {
   449→   449→        e.preventDefault();
   450→   450→        setAtIndex((i) => (i + 1) % atFiltered.length);
   451→   451→        return;
   452→   452→      }
   453→   453→      if (e.key === "ArrowUp") {
   454→   454→        e.preventDefault();
   455→   455→        setAtIndex((i) => (i - 1 + atFiltered.length) % atFiltered.length);
   456→   456→        return;
   457→   457→      }
   458→   458→      if (e.key === "Enter" || e.key === "Tab") {
   459→   459→        e.preventDefault();
   460→   460→        const item = atFiltered[atIndex] ?? atFiltered[0];
   461→   461→        if (item) selectRef(item);
   462→   462→        return;
   463→   463→      }
   464→   464→      if (e.key === "Escape") {
   465→   465→        e.preventDefault();
   466→   466→        setAtSuppressed(true);
   467→   467→        return;
   468→   468→      }
   469→   469→    }
   470→   470→    if (e.key === "Enter" && !e.shiftKey && !e.nativeEvent.isComposing) {
   471→   471→      if (!streaming && draft.trim()) {
   472→   472→        e.preventDefault();
   473→   473→        handleSend();
   474→   474→      }
   475→   475→    }
   476→   476→  };
   477→   477→
   478→   478→  /* ---- voice ---- */
   479→   479→  const startVoice = React.useCallback(() => {
   480→   480→    const w = window as WindowWithSpeech;
   481→   481→    const Ctor = w.SpeechRecognition ?? w.webkitSpeechRecognition;
   482→   482→    if (!Ctor) {
   483→   483→      toast.error("Voice input not supported in this browser");
   484→   484→      return;
   485→   485→    }
   486→   486→    try {
   487→   487→      const rec = new Ctor();
   488→   488→      rec.lang = "en-US";
   489→   489→      rec.continuous = false;
   490→   490→      rec.interimResults = false;
