    79→function ProviderCard({
    80→  provider,
    81→  onRemove,
    82→}: {
    83→  provider: Provider;
    84→  onRemove: (p: Provider) => void;
    85→}) {
    86→  const updateProvider = useProvidersStore((s) => s.updateProvider);
    87→  const toggleModelHidden = useProvidersStore((s) => s.toggleModelHidden);
    88→  const detectOllama = useProvidersStore((s) => s.detectOllama);
    89→  const detecting = useProvidersStore((s) => s.detecting);
    90→  const isOllama = provider.type === "ollama";
    91→
    92→  const onDetect = async () => {
    93→    try {
    94→      const models = await detectOllama(provider.baseUrl);
    95→      if (models.length === 0) {
    96→        toast.error("Ollama not reachable");
    97→      } else {
    98→        toast.success(`${models.length} Ollama model(s) detected`);
    99→      }
   100→    } catch {
   101→      toast.error("Ollama not reachable");
   102→    }
   103→  };
   104→
   105→  return (
   106→    <div className="rounded-[10px] border border-ws-border bg-card-ws p-3">
   107→      <div className="flex items-center gap-2">
   108→        <Input
   109→          aria-label="Provider name"
   110→          value={provider.name}
   111→          onChange={(e) =>
   112→            updateProvider(provider.id, { name: e.target.value })
   113→          }
   114→          className="flex-1 h-8 bg-card-ws border-ws-border-soft text-[13px] focus-ring"
   115→        />
   116→        <span className="text-[11px] rounded-full px-1.5 py-0.5 border text-ws-secondary border-ws-border whitespace-nowrap">
   117→          {PROVIDER_TYPE_LABEL[provider.type]}
   118→        </span>
   119→        <Switch
   120→          checked={provider.enabled}
   121→          onCheckedChange={(v) =>
   122→            updateProvider(provider.id, { enabled: v })
   123→          }
   124→          aria-label="Toggle provider enabled"
   125→        />
   126→        <button
   127→          type="button"
   128→          onClick={() => onRemove(provider)}
   129→          aria-label={`Remove ${provider.name}`}
   130→          className="inline-flex items-center justify-center size-7 text-ws-secondary hover:text-red-400 hover:bg-hover rounded-[8px] focus-ring"
   131→        >
   132→          <Trash2 className="size-3.5" />
   133→        </button>
   134→      </div>
   135→
   136→      <div className="mt-2 grid gap-2">
   137→        <FieldRow label="Base URL" htmlFor={`prov-base-${provider.id}`}>
   138→          <Input
   139→            id={`prov-base-${provider.id}`}
   140→            value={provider.baseUrl}
   141→            onChange={(e) =>
   142→              updateProvider(provider.id, { baseUrl: e.target.value })
   143→            }
   144→            placeholder="https://api.openai.com/v1"
   145→            className="h-8 bg-card-ws border-ws-border-soft text-[13px] focus-ring"
   146→          />
   147→        </FieldRow>
   148→
   149→        {isOllama ? (
   150→          <p className="text-[12px] text-ws-secondary px-1 py-1">
   151→            Ollama runs locally and needs no API key. Use the detect button
   152→            below to fetch available models.
   153→          </p>
   154→        ) : (
   155→          <FieldRow label="API key" htmlFor={`prov-key-${provider.id}`}>
   156→            <Input
   157→              id={`prov-key-${provider.id}`}
   158→              type="password"
   159→              value={provider.apiKey}
   160→              onChange={(e) =>
   161→                updateProvider(provider.id, { apiKey: e.target.value })
   162→              }
   163→              placeholder="sk-..."
   164→              className="h-8 bg-card-ws border-ws-border-soft text-[13px] focus-ring"
   165→            />
   166→          </FieldRow>
   167→        )}
   168→      </div>
   169→
   170→      <div className="mt-2 flex flex-col gap-0.5">
   171→        <div className="flex items-center justify-between px-2">
   172→          <span className="text-[11px] uppercase tracking-wide text-ws-secondary">
   173→            Models
   174→          </span>
   175→          <span className="text-[11px] text-ws-secondary">
   176→            {provider.models.length}
   177→          </span>
   178→        </div>
   179→
   180→        {provider.models.length === 0 ? (
   181→          <div className="flex items-center justify-between px-2">
   182→            <span className="text-[12px] text-ws-disabled py-1">No models</span>
   183→            {isOllama ? (
   184→              <button
   185→                type="button"
   186→                onClick={onDetect}
   187→                disabled={detecting}
   188→                className="inline-flex items-center gap-2 h-8 px-3 rounded-[8px] bg-accent-fill text-white text-[13px] hover:opacity-90 focus-ring disabled:opacity-60"
   189→              >
   190→                {detecting ? (
   191→                  <Loader2 className="size-3.5 animate-spin" />
   192→                ) : (
   193→                  <RefreshCw className="size-3.5" />
   194→                )}
   195→                {detecting ? "Detecting..." : "Detect models"}
   196→              </button>
   197→            ) : null}
   198→          </div>
   199→        ) : (
   200→          <>
   201→            {provider.models.map((m) => (
   202→              <div
   203→                key={m.id}
   204→                className="flex items-center gap-2 rounded-[6px] px-2 py-1 hover:bg-hover"
   205→              >
   206→                <span className="text-[12.5px] text-foreground font-mono truncate">
   207→                  {m.name}
   208→                </span>
   209→                {m.hidden ? (
   210→                  <span className="text-[10px] text-ws-disabled border border-ws-border rounded-full px-1.5 py-0.5">
   211→                    hidden
   212→                  </span>
   213→                ) : null}
   214→                <button
   215→                  type="button"
   216→                  onClick={() => toggleModelHidden(provider.id, m.id)}
   217→                  aria-label={m.hidden ? "Show model" : "Hide model"}
   218→                  className="ml-auto inline-flex items-center justify-center size-6 rounded-[6px] text-ws-secondary hover:bg-card-ws hover:text-foreground focus-ring"
   219→                >
   220→                  {m.hidden ? (
   221→                    <EyeOff className="size-3.5" />
   222→                  ) : (
   223→                    <Eye className="size-3.5" />
   224→                  )}
   225→                </button>
   226→              </div>
   227→            ))}
   228→            {isOllama ? (
   229→              <div className="flex justify-end mt-1">
   230→                <button
   231→                  type="button"
   232→                  onClick={onDetect}
   233→                  disabled={detecting}
   234→                  className="inline-flex items-center gap-2 h-8 px-3 rounded-[8px] bg-accent-fill text-white text-[13px] hover:opacity-90 focus-ring disabled:opacity-60"
   235→                >
   236→                  {detecting ? (
   237→                    <Loader2 className="size-3.5 animate-spin" />
   238→                  ) : (
   239→                    <RefreshCw className="size-3.5" />
   240→                  )}
   241→                  {detecting ? "Detecting..." : "Detect models"}
   242→                </button>
   243→              </div>
   244→            ) : null}
   245→          </>
   246→        )}
   247→      </div>
   248→    </div>
   249→  );
   250→}

   252→export function SettingsModal() {
   253→  const open = useAppStore((s) => s.settingsOpen);
   254→  const setOpen = useAppStore((s) => s.setSettingsOpen);
   255→
   256→  const apiKey = useSettingsStore((s) => s.apiKey);
   257→  const setApiKey = useSettingsStore((s) => s.setApiKey);
   258→  const baseUrl = useSettingsStore((s) => s.baseUrl);
   259→  const setBaseUrl = useSettingsStore((s) => s.setBaseUrl);
   260→  const defaultModelId = useSettingsStore((s) => s.defaultModelId);
   261→  const setDefaultModelId = useSettingsStore((s) => s.setDefaultModelId);
   262→  const theme = useSettingsStore((s) => s.theme);
   263→  const setTheme = useSettingsStore((s) => s.setTheme);
   264→  const agentInstructions = useSettingsStore((s) => s.agentInstructions);
   265→  const setAgentInstructions = useSettingsStore((s) => s.setAgentInstructions);
   266→  const skills = useSettingsStore((s) => s.skills);
   267→  const addSkill = useSettingsStore((s) => s.addSkill);
   268→  const removeSkill = useSettingsStore((s) => s.removeSkill);
   269→  const skillsFolderPath = useSettingsStore((s) => s.skillsFolderPath);
   270→  const setSkillsFolderPath = useSettingsStore((s) => s.setSkillsFolderPath);
   271→  const useRealApi = useSettingsStore((s) => s.useRealApi);
   272→  const setUseRealApi = useSettingsStore((s) => s.setUseRealApi);
   273→
   274→  const providers = useProvidersStore((s) => s.providers);
   275→  const addProvider = useProvidersStore((s) => s.addProvider);
   276→  const removeProvider = useProvidersStore((s) => s.removeProvider);
   277→
   278→  const [newSkill, setNewSkill] = React.useState("");
   279→  const [tab, setTab] = React.useState("general");
   280→
   281→  const [addOpen, setAddOpen] = React.useState(false);
   282→  const [newName, setNewName] = React.useState("");
   283→  const [newType, setNewType] = React.useState<ProviderType>("custom");
   284→  const [newBaseUrl, setNewBaseUrl] = React.useState("");
   285→  const [removeTarget, setRemoveTarget] = React.useState<Provider | null>(
   286→    null
   287→  );
   288→
   289→  const openAddProvider = () => {
   290→    setNewName("");
   291→    setNewType("custom");
   292→    setNewBaseUrl("");
   293→    setAddOpen(true);
   294→  };
   295→
   296→  const confirmAddProvider = () => {
   297→    const name = newName.trim();
   298→    if (!name) {
   299→      toast.error("Provider name is required");
   300→      return;
   301→    }
   302→    addProvider({ name, type: newType, baseUrl: newBaseUrl.trim() });
   303→    toast.success("Provider added");
   304→    setAddOpen(false);
   305→  };
   306→
   307→  const confirmRemove = () => {
   308→    if (!removeTarget) return;
   309→    removeProvider(removeTarget.id);
   310→    toast.success("Provider removed");
   311→    setRemoveTarget(null);
   312→  };
   313→
   314→  const addSkillOnEnter = (e: React.KeyboardEvent<HTMLInputElement>) => {
   315→    if (e.key !== "Enter") return;
   316→    e.preventDefault();
   317→    const v = newSkill.trim();
   318→    if (!v) return;
   319→    addSkill(v);
   320→    setNewSkill("");
   321→  };
   322→
   323→  const folderInputRef = React.useRef<HTMLInputElement>(null);
   324→
   325→  React.useEffect(() => {
   326→    const el = folderInputRef.current;
   327→    if (el) {
   328→      el.setAttribute("webkitdirectory", "");
   329→    }
   330→  }, []);
   331→
   332→  const onFolderPicked = (e: React.ChangeEvent<HTMLInputElement>) => {
   333→    const fileList = e.target.files;
   334→    if (!fileList || fileList.length === 0) {
   335→      e.target.value = "";
   336→      return;
   337→    }
   338→    const first = fileList[0] as File & { webkitRelativePath?: string };
   339→    const rel = first.webkitRelativePath;
   340→    if (rel) {
   341→      const folderName = rel.split("/")[0];
   342→      if (folderName) setSkillsFolderPath(folderName);
   343→    }
   344→    const added = new Set<string>();
   345→    for (let i = 0; i < fileList.length; i++) {
   346→      const name = fileList[i].name;
   347→      const lower = name.toLowerCase();
   348→      if (!lower.endsWith(".md") && !lower.endsWith(".txt")) continue;
   349→      const base = name.replace(/\.[^.]+$/, "");
   350→      if (!base) continue;
   351→      if (skills.includes(base) || added.has(base)) continue;
   352→      addSkill(base);
   353→      added.add(base);
   354→    }
   355→    if (added.size > 0) {
   356→      toast.success(`${added.size} skill(s) imported from folder`);
   357→    } else {
   358→      toast("No skill files found in folder");
   359→    }
   360→    e.target.value = "";
   361→  };
   362→
   363→  return (
   364→    <>
   365→    <Dialog open={open} onOpenChange={setOpen}>
   366→      <DialogContent className="max-w-[640px] p-0 bg-popover border border-ws-border">

   368→          <DialogTitle className="text-[15px]">Settings</DialogTitle>

   370→            Configure API key, models, agent instructions and skills.
   371→          </DialogDescription>
   372→        </DialogHeader>
   373→
   374→        <div className="px-5 py-4">
   375→          <Tabs value={tab} onValueChange={setTab}>
   376→            <TabsList className="bg-card-ws border border-ws-border h-8">
   377→              <TabsTrigger value="general" className="text-[12.5px]">
   378→                General
   379→              </TabsTrigger>
   380→              <TabsTrigger value="providers" className="text-[12.5px]">
   381→                Providers
   382→              </TabsTrigger>
   383→              <TabsTrigger value="models" className="text-[12.5px]">
   384→                Models
   385→              </TabsTrigger>
   386→              <TabsTrigger value="agents" className="text-[12.5px]">
   387→                Agents
   388→              </TabsTrigger>
   389→              <TabsTrigger value="skills" className="text-[12.5px]">
   390→                Skills
   391→              </TabsTrigger>
   392→            </TabsList>
   393→
   394→            <TabsContent value="general" className="mt-4">
   395→              <div className="grid gap-4">
   396→                <FieldRow label="API key" htmlFor="settings-api-key">
   397→                  <Input
   398→                    id="settings-api-key"
   399→                    type="password"
   400→                    value={apiKey}
   401→                    onChange={(e) => setApiKey(e.target.value)}
   402→                    placeholder="sk-..."
   403→                    className="focus-ring bg-card-ws border-ws-border text-[13px]"

   405→                </FieldRow>

   408→                    id="settings-base-url"
   409→                    value={baseUrl}
   410→                    onChange={(e) => setBaseUrl(e.target.value)}
   411→                    placeholder="https://api.openai.com/v1"
   412→                    className="focus-ring bg-card-ws border-ws-border text-[13px]"

   414→                </FieldRow>
