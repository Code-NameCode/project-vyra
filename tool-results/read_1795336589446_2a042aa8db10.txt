    77→/** A provider is connected when enabled and either ollama (no key needed) or has an API key. */

    82→function ProviderCard({
    83→  provider,
    84→  onRemove,
    85→}: {
    86→  provider: Provider;
    87→  onRemove: (p: Provider) => void;
    88→}) {
    89→  const updateProvider = useProvidersStore((s) => s.updateProvider);
    90→  const detectProviderModels = useProvidersStore((s) => s.detectProviderModels);
    91→  const isOllama = provider.type === "ollama";
    92→  const connected = isConnected(provider);
    93→  const [fetching, setFetching] = React.useState(false);
    94→
    95→  const onFetch = async () => {
    96→    if (!connected) return;
    97→    setFetching(true);
    98→    try {
    99→      const models = await detectProviderModels(provider.id);
   100→      if (models.length > 0) {
   101→        toast.success(
   102→          `${models.length} model(s) fetched from ${provider.name}`
   103→        );
   104→      } else {
   105→        toast.error("No models found — check the connection");
   106→      }
   107→    } catch {
   108→      toast.error("No models found — check the connection");
   109→    } finally {
   110→      setFetching(false);
   111→    }
   112→  };
   113→
   114→  return (
   115→    <div className="rounded-[10px] border border-ws-border bg-card-ws p-3">
   116→      <div className="flex items-center gap-2">
   117→        <Input
   118→          aria-label="Provider name"
   119→          value={provider.name}
   120→          onChange={(e) =>
   121→            updateProvider(provider.id, { name: e.target.value })
   122→          }
   123→          className="flex-1 h-8 bg-card-ws border-ws-border-soft text-[13px] focus-ring"
   124→        />
   125→        <span className="text-[11px] rounded-full px-1.5 py-0.5 border text-ws-secondary border-ws-border whitespace-nowrap">
   126→          {PROVIDER_TYPE_LABEL[provider.type]}
   127→        </span>
   128→        <Switch
   129→          checked={provider.enabled}
   130→          onCheckedChange={(v) =>
   131→            updateProvider(provider.id, { enabled: v })
   132→          }
   133→          aria-label="Toggle provider enabled"
   134→        />
   135→        <button
   136→          type="button"
   137→          onClick={() => onRemove(provider)}
   138→          aria-label={`Remove ${provider.name}`}
   139→          className="inline-flex items-center justify-center size-7 text-ws-secondary hover:text-red-400 hover:bg-hover rounded-[8px] focus-ring"
   140→        >
   141→          <Trash2 className="size-3.5" />
   142→        </button>
   143→      </div>
   144→
   145→      <div className="mt-2 grid gap-2">
   146→        <FieldRow label="Base URL" htmlFor={`prov-base-${provider.id}`}>
   147→          <Input
   148→            id={`prov-base-${provider.id}`}
   149→            value={provider.baseUrl}
   150→            onChange={(e) =>
   151→              updateProvider(provider.id, { baseUrl: e.target.value })
   152→            }
   153→            placeholder="https://api.openai.com/v1"
   154→            className="h-8 bg-card-ws border-ws-border-soft text-[13px] focus-ring"
   155→          />
   156→        </FieldRow>
   157→
   158→        {isOllama ? (
   159→          <p className="text-[12px] text-ws-secondary px-1 py-1">
   160→            Ollama runs locally and needs no API key. Use Fetch models to load
   161→            available models.
   162→          </p>
   163→        ) : (
   164→          <FieldRow label="API key" htmlFor={`prov-key-${provider.id}`}>
   165→            <Input
   166→              id={`prov-key-${provider.id}`}
   167→              type="password"
   168→              value={provider.apiKey}
   169→              onChange={(e) =>
   170→                updateProvider(provider.id, { apiKey: e.target.value })
   171→              }
   172→              placeholder="sk-..."
   173→              className="h-8 bg-card-ws border-ws-border-soft text-[13px] focus-ring"
   174→            />
   175→          </FieldRow>
   176→        )}
   177→      </div>
   178→
   179→      <div className="mt-2 flex items-center justify-between px-2">
   180→        <span className="text-[12px] text-ws-secondary">
   181→          {provider.models.length} model(s)
   182→        </span>
   183→        <button
   184→          type="button"
   185→          onClick={onFetch}
   186→          disabled={!connected || fetching}
   187→          className={cn(
   188→            "inline-flex items-center gap-2 h-7 px-2.5 rounded-[8px] bg-accent-fill text-white text-[12px] hover:opacity-90 focus-ring",
   189→            (!connected || fetching) && "opacity-50 cursor-not-allowed"
   190→          )}
   191→        >
   192→          {fetching ? (
   193→            <Loader2 className="size-3.5 animate-spin" />
   194→          ) : (
   195→            <RefreshCw className="size-3.5" />
   196→          )}
   197→          {fetching ? "Fetching…" : "Fetch models"}
   198→        </button>
   199→      </div>
   200→    </div>
   201→  );
   202→}

   204→function ModelsTabContent() {

   216→        toast.success(`${models.length} model(s) fetched from ${p.name}`);

   218→        toast.error("No models found — check the connection");
   219→      }
   220→    } catch {
   221→      toast.error("No models found — check the connection");

   276→                      "inline-flex items-center gap-2 h-7 px-2.5 rounded-[8px] bg-accent-fill text-white text-[12px] hover:opacity-90 focus-ring",
   277→                      (!connected || fetching) &&
   278→                        "opacity-50 cursor-not-allowed"
   279→                    )}
   280→                  >
   281→                    {fetching ? (
   282→                      <Loader2 className="size-3.5 animate-spin" />

   284→                      <RefreshCw className="size-3.5" />
   285→                    )}
   286→                    {fetching ? "Fetching…" : "Fetch models"}

   296→                    p.models.map((m) => (

   298→                        key={m.id}

   300→                      >
   301→                        <span className="text-[12.5px] text-foreground font-mono truncate">
   302→                          {m.name}
   303→                        </span>

   305→                          <span className="text-[10px] text-ws-disabled border border-ws-border rounded-full px-1.5 py-0.5">
   306→                            hidden
   307→                          </span>
   308→                        ) : null}
   309→                        <button
   310→                          type="button"
   311→                          onClick={() => toggleModelHidden(p.id, m.id)}
   312→                          aria-label={m.hidden ? "Show" : "Hide"}

   325→              </div>
   326→            );
   327→          })

   329→      </div>

   334→export function SettingsModal() {
   335→  const open = useAppStore((s) => s.settingsOpen);
   336→  const setOpen = useAppStore((s) => s.setSettingsOpen);
   337→
   338→  const apiKey = useSettingsStore((s) => s.apiKey);
   339→  const setApiKey = useSettingsStore((s) => s.setApiKey);
   340→  const baseUrl = useSettingsStore((s) => s.baseUrl);
   341→  const setBaseUrl = useSettingsStore((s) => s.setBaseUrl);
   342→  const defaultModelId = useSettingsStore((s) => s.defaultModelId);
   343→  const setDefaultModelId = useSettingsStore((s) => s.setDefaultModelId);
   344→  const theme = useSettingsStore((s) => s.theme);
   345→  const setTheme = useSettingsStore((s) => s.setTheme);
   346→  const agentInstructions = useSettingsStore((s) => s.agentInstructions);
   347→  const setAgentInstructions = useSettingsStore((s) => s.setAgentInstructions);
   348→  const skills = useSettingsStore((s) => s.skills);
   349→  const addSkill = useSettingsStore((s) => s.addSkill);
   350→  const removeSkill = useSettingsStore((s) => s.removeSkill);
   351→  const skillsFolderPath = useSettingsStore((s) => s.skillsFolderPath);
   352→  const setSkillsFolderPath = useSettingsStore((s) => s.setSkillsFolderPath);
   353→  const useRealApi = useSettingsStore((s) => s.useRealApi);
   354→  const setUseRealApi = useSettingsStore((s) => s.setUseRealApi);
   355→
   356→  const providers = useProvidersStore((s) => s.providers);
   357→  const addProvider = useProvidersStore((s) => s.addProvider);
   358→  const removeProvider = useProvidersStore((s) => s.removeProvider);
   359→
   360→  // Visible provider models (from connected providers, non-hidden) — used to
   361→  // populate the General tab's default-model picker instead of the old static
   362→  // built-in model list.
   363→  const visibleModels = React.useMemo(() => {
   364→    const out: { id: string; label: string }[] = [];
   365→    for (const p of providers) {
   366→      if (!isConnected(p)) continue;
   367→      for (const m of p.models) {
   368→        if (m.hidden) continue;
   369→        out.push({ id: m.id, label: m.name });
   370→      }
   371→    }
   372→    return out;
   373→  }, [providers]);
   374→
   375→  const [newSkill, setNewSkill] = React.useState("");
   376→  const [tab, setTab] = React.useState("general");
   377→
   378→  const [addOpen, setAddOpen] = React.useState(false);
   379→  const [newName, setNewName] = React.useState("");
   380→  const [newType, setNewType] = React.useState<ProviderType>("custom");
   381→  const [newBaseUrl, setNewBaseUrl] = React.useState("");
   382→  const [removeTarget, setRemoveTarget] = React.useState<Provider | null>(
   383→    null
   384→  );
   385→
   386→  const openAddProvider = () => {
   387→    setNewName("");
   388→    setNewType("custom");
   389→    setNewBaseUrl("");
   390→    setAddOpen(true);
   391→  };
   392→
   393→  const confirmAddProvider = () => {
   394→    const name = newName.trim();
   395→    if (!name) {
   396→      toast.error("Provider name is required");
   397→      return;
   398→    }
   399→    addProvider({ name, type: newType, baseUrl: newBaseUrl.trim() });
   400→    toast.success("Provider added");
   401→    setAddOpen(false);
   402→  };
   403→
   404→  const confirmRemove = () => {
   405→    if (!removeTarget) return;
   406→    removeProvider(removeTarget.id);
   407→    toast.success("Provider removed");
   408→    setRemoveTarget(null);
   409→  };
   410→
   411→  const addSkillOnEnter = (e: React.KeyboardEvent<HTMLInputElement>) => {
   412→    if (e.key !== "Enter") return;
   413→    e.preventDefault();
   414→    const v = newSkill.trim();
   415→    if (!v) return;
   416→    addSkill(v);
   417→    setNewSkill("");
   418→  };
   419→
   420→  const folderInputRef = React.useRef<HTMLInputElement>(null);
   421→
   422→  React.useEffect(() => {
   423→    const el = folderInputRef.current;
   424→    if (el) {
   425→      el.setAttribute("webkitdirectory", "");
   426→    }
   427→  }, []);
   428→
   429→  const onFolderPicked = (e: React.ChangeEvent<HTMLInputElement>) => {
   430→    const fileList = e.target.files;
   431→    if (!fileList || fileList.length === 0) {
   432→      e.target.value = "";
   433→      return;
   434→    }
   435→    const first = fileList[0] as File & { webkitRelativePath?: string };
   436→    const rel = first.webkitRelativePath;
   437→    if (rel) {
   438→      const folderName = rel.split("/")[0];
   439→      if (folderName) setSkillsFolderPath(folderName);
   440→    }
   441→    const added = new Set<string>();
   442→    for (let i = 0; i < fileList.length; i++) {
   443→      const name = fileList[i].name;
   444→      const lower = name.toLowerCase();
   445→      if (!lower.endsWith(".md") && !lower.endsWith(".txt")) continue;
   446→      const base = name.replace(/\.[^.]+$/, "");
   447→      if (!base) continue;
   448→      if (skills.includes(base) || added.has(base)) continue;
   449→      addSkill(base);
   450→      added.add(base);
   451→    }
   452→    if (added.size > 0) {
   453→      toast.success(`${added.size} skill(s) imported from folder`);
   454→    } else {
   455→      toast("No skill files found in folder");
   456→    }
   457→    e.target.value = "";
   458→  };
   459→
   460→  return (
   461→    <>
   462→    <Dialog open={open} onOpenChange={setOpen}>
   463→      <DialogContent className="max-w-[640px] p-0 bg-popover border border-ws-border">
   464→        <DialogHeader className="px-5 pt-5 pb-3 border-b border-ws-border">
   465→          <DialogTitle className="text-[15px]">Settings</DialogTitle>
   466→          <DialogDescription className="sr-only">
   467→            Configure API key, models, agent instructions and skills.
   468→          </DialogDescription>
   469→        </DialogHeader>
   470→
   471→        <div className="px-5 py-4">
   472→          <Tabs value={tab} onValueChange={setTab}>
   473→            <TabsList className="bg-card-ws border border-ws-border h-8">
   474→              <TabsTrigger value="general" className="text-[12.5px]">
   475→                General
   476→              </TabsTrigger>
   477→              <TabsTrigger value="providers" className="text-[12.5px]">
   478→                Providers
   479→              </TabsTrigger>
   480→              <TabsTrigger value="models" className="text-[12.5px]">
   481→                Models
   482→              </TabsTrigger>
   483→              <TabsTrigger value="agents" className="text-[12.5px]">
   484→                Agents
   485→              </TabsTrigger>
   486→              <TabsTrigger value="skills" className="text-[12.5px]">
   487→                Skills
   488→              </TabsTrigger>
   489→            </TabsList>
   490→
   491→            <TabsContent value="general" className="mt-4">
   492→              <div className="grid gap-4">
   493→                <FieldRow label="API key" htmlFor="settings-api-key">
   494→                  <Input
   495→                    id="settings-api-key"
   496→                    type="password"
   497→                    value={apiKey}
   498→                    onChange={(e) => setApiKey(e.target.value)}
   499→                    placeholder="sk-..."
   500→                    className="focus-ring bg-card-ws border-ws-border text-[13px]"
   501→                  />
   502→                </FieldRow>
   503→                <FieldRow label="Base URL" htmlFor="settings-base-url">
   504→                  <Input
   505→                    id="settings-base-url"
   506→                    value={baseUrl}
   507→                    onChange={(e) => setBaseUrl(e.target.value)}
   508→                    placeholder="https://api.openai.com/v1"
   509→                    className="focus-ring bg-card-ws border-ws-border text-[13px]"
   510→                  />
   511→                </FieldRow>
   512→                <div className="flex items-center justify-between rounded-lg border border-ws-border bg-card-ws px-3 py-2.5">
   513→                  <div className="grid gap-0.5">
   514→                    <span className="text-[13px] text-foreground">
   515→                      Use real OpenAI-compatible API
   516→                    </span>
   517→                    <span className="text-[12px] text-ws-disabled">
   518→                      When off, Goodstuff-Agent uses mock streaming responses.
   519→                    </span>
   520→                  </div>
   521→                  <Switch
   522→                    checked={useRealApi}
   523→                    onCheckedChange={setUseRealApi}
   524→                    aria-label="Use real API"
   525→                  />
   526→                </div>
   527→                <FieldRow label="Default model" htmlFor="settings-model">
   528→                  <Select
   529→                    value={
   530→                      visibleModels.some((m) => m.id === defaultModelId)
   531→                        ? defaultModelId
   532→                        : undefined
   533→                    }
   534→                    onValueChange={setDefaultModelId}
   535→                  >
   536→                    <SelectTrigger
   537→                      id="settings-model"
   538→                      className="focus-ring bg-card-ws border-ws-border text-[13px] w-full"
   539→                    >
   540→                      <SelectValue placeholder="Select default model" />
   541→                    </SelectTrigger>
   542→                    <SelectContent className="bg-popover border-ws-border">
   543→                      {visibleModels.length === 0 ? (
   544→                        <SelectItem
   545→                          value="__none__"
   546→                          disabled
   547→                          className="text-[13px]"
   548→                        >
   549→                          No models available
   550→                        </SelectItem>
   551→                      ) : (
   552→                        visibleModels.map((m) => (
   553→                          <SelectItem
   554→                            key={m.id}
   555→                            value={m.id}
   556→                            className="text-[13px]"
   557→                          >
   558→                            {m.label}
   559→                          </SelectItem>
   560→                        ))
   561→                      )}
   562→                    </SelectContent>
   563→                  </Select>
   564→                </FieldRow>
   565→                <FieldRow label="Theme" htmlFor="settings-theme" hint="Goodstuff-Agent is always dark visually.">
   566→                  <Select value={theme} onValueChange={(v) => setTheme(v as "dark" | "light")}>
   567→                    <SelectTrigger
   568→                      id="settings-theme"
   569→                      className="focus-ring bg-card-ws border-ws-border text-[13px] w-full"
   570→                    >
   571→                      <SelectValue placeholder="Select theme" />
   572→                    </SelectTrigger>
   573→                    <SelectContent className="bg-popover border-ws-border">
   574→                      <SelectItem value="dark" className="text-[13px]">
   575→                        Dark
   576→                      </SelectItem>
