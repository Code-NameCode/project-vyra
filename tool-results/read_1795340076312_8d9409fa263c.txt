    67→/* ---- minimal Web Speech API typing (no `any`) ---- */
    68→interface SpeechResultLike {
    69→  0: { transcript: string };
    70→  isFinal: boolean;
    71→  length: number;
    72→}
    73→interface SpeechEventLike {
    74→  results: ArrayLike<SpeechResultLike>;
    75→}
    76→interface SpeechRecognitionLike {
    77→  lang: string;
    78→  continuous: boolean;
    79→  interimResults: boolean;
    80→  start: () => void;
    81→  stop: () => void;
    82→  abort: () => void;
    83→  onresult: ((event: SpeechEventLike) => void) | null;
    84→  onerror: ((event: unknown) => void) | null;
    85→  onend: (() => void) | null;
    86→}
    87→type SpeechRecognitionCtor = new () => SpeechRecognitionLike;
    88→type WindowWithSpeech = Window & {
    89→  SpeechRecognition?: SpeechRecognitionCtor;
    90→  webkitSpeechRecognition?: SpeechRecognitionCtor;
    91→};

    93→/* ---- Plan / Build mode ---- */
    94→type BuildMode = "plan" | "build";
    95→// Other modules may extend Conversation/ChatState with buildMode. Access
    96→// defensively so the Composer keeps working whether or not the store has
    97→// been augmented yet.
    98→type ConvWithBuild = Conversation & { buildMode?: BuildMode };
    99→interface ChatStoreWithBuild {
   100→  setBuildMode?: (tabId: string, mode: BuildMode) => void;
   101→}

   139→/* ---- @ context reference item ---- */
   140→type SlashCommandWithIcon = SlashCommand & { icon?: string };
   141→interface RefItem {
   142→  kind: "file" | "project";
   143→  name: string;
   144→  path: string;
   145→  icon: LucideIcon;
   146→}
   147→const EMPTY_FILES: FileNode[] = [];
   148→const EMPTY_SKILLS: string[] = [];
   149→
   150→/** Strip a leading `/` from a slash command label (e.g. "/review" -> "review"). */
   151→function stripSlash(label: string): string {
   152→  return label.replace(/^\//, "");
   153→}

   155→/**
   156→ * Composer — the core prompt input.
   157→ * Auto-growing textarea, slash command popup (with skill icons) that adds
   158→ * action skills as removable chips, Codex-style `@` context referencing,
   159→ * attach menu, Plan/Build mode selector, agent picker chip, model picker
   160→ * chip, and a send/stop/voice button.
   161→ */
   162→export function Composer({ tabId }: { tabId: string }) {
   163→  const ensure = useChatStore((s) => s.ensure);
   164→  const draft = useChatStore((s) => s.conversations[tabId]?.composerDraft ?? "");
   165→  const streaming = useChatStore(
   166→    (s) => s.conversations[tabId]?.streaming ?? false
   167→  );
   168→  const convModelId = useChatStore((s) => s.conversations[tabId]?.modelId);
   169→  const defaultModelId = useSettingsStore((s) => s.defaultModelId);
   170→  const setComposerDraft = useChatStore((s) => s.setComposerDraft);
   171→  const sendMessage = useChatStore((s) => s.sendMessage);
   172→  const stopStreaming = useChatStore((s) => s.stopStreaming);
   173→  const clearConversation = useChatStore((s) => s.clearConversation);
   174→
   175→  /* ---- skills (multi-chip) ---- */
   176→  const skills = useChatStore(
   177→    (s) => s.conversations[tabId]?.skills ?? EMPTY_SKILLS
   178→  );
   179→  const addSkill = useChatStore((s) => s.addSkill);
   180→  const removeSkill = useChatStore((s) => s.removeSkill);
   181→
   182→  /* ---- agent picker ---- */
   183→  const agentId = useChatStore(
   184→    (s) => s.conversations[tabId]?.agentId ?? ""
   185→  );
   186→  const setAgent = useChatStore((s) => s.setAgent);
   187→
   188→  /* ---- Plan / Build mode (defensive: works with or without store support) ---- */
   189→  const storeBuildMode = useChatStore(
   190→    (s) =>
   191→      (s.conversations[tabId] as ConvWithBuild | undefined)?.buildMode ??
   192→      ("build" as BuildMode)
   193→  );
   194→  const storeSetBuildMode = useChatStore(
   195→    (s) => (s as unknown as ChatStoreWithBuild).setBuildMode
   196→  );
   197→  const [localBuildMode, setLocalBuildMode] =
   198→    React.useState<BuildMode>("build");
   199→  const buildMode: BuildMode = storeSetBuildMode
   200→    ? (storeBuildMode as BuildMode)
   201→    : localBuildMode;
   202→  const setBuildMode = React.useCallback(
   203→    (mode: BuildMode) => {
   204→      if (storeSetBuildMode) storeSetBuildMode(tabId, mode);
   205→      else setLocalBuildMode(mode);
   206→    },
   207→    [storeSetBuildMode, tabId]
   208→  );
   209→
   210→  /* ---- active workspace (reactive) for `@` referencing ---- */
   211→  const projectId = useAppStore(
   212→    (s) => s.tabs.find((t) => t.id === tabId)?.projectId
   213→  );
   214→  const projects = useProjectsStore((s) => s.projects);
   215→  const setTabProject = useAppStore((s) => s.setTabProject);
   216→  const addProject = useProjectsStore((s) => s.addProject);
   217→  const activeProject = projects.find((p) => p.id === projectId);
   218→  const convFiles = useChatStore(
   219→    (s) => s.conversations[tabId]?.workspace.files
   220→  );
   221→
   222→  const files = React.useMemo<FileNode[]>(() => {
   223→    if (projectId) {
   224→      return (
   225→        projects.find((p) => p.id === projectId)?.workspace.files ??
   226→        EMPTY_FILES
   227→      );
   228→    }
   229→    return convFiles ?? EMPTY_FILES;
   230→  }, [projectId, projects, convFiles]);
   231→
   232→  const refItems = React.useMemo<RefItem[]>(() => {
   233→    const fileItems: RefItem[] = files.map((f) => ({
   234→      kind: "file",
   235→      name: f.name,
   236→      path: f.path,
   237→      icon: File,
   238→    }));
   239→    const projItems: RefItem[] = projects.map((p) => ({
   240→      kind: "project",
   241→      name: p.name,
   242→      path: p.name,
   243→      icon: Folder,
   244→    }));
   245→    return [...fileItems, ...projItems];
   246→  }, [files, projects]);
   247→
   248→  const modelId = convModelId ?? defaultModelId;
   249→
   250→  const taRef = React.useRef<HTMLTextAreaElement>(null);
   251→  const [modelOpen, setModelOpen] = React.useState(false);
   252→  const [agentOpen, setAgentOpen] = React.useState(false);
   253→  const [slashSuppressed, setSlashSuppressed] = React.useState(false);
   254→  const [selIndex, setSelIndex] = React.useState(0);
   255→  const [atSuppressed, setAtSuppressed] = React.useState(false);
   256→  const [atIndex, setAtIndex] = React.useState(0);
   257→
   258→  // ensure a conversation object exists so modelId/draft are stable
   259→  React.useEffect(() => {
   260→    ensure(tabId);
   261→  }, [tabId, ensure]);
   262→
   263→  // auto-grow textarea whenever draft changes (covers typing + external updates)
   264→  React.useLayoutEffect(() => {
   265→    const ta = taRef.current;
   266→    if (!ta) return;
   267→    ta.style.height = "auto";
   268→    ta.style.height = `${Math.min(ta.scrollHeight, 200)}px`;
   269→  }, [draft]);
   270→
   271→  // reset slash + @ suppression + selection whenever draft changes
   272→  React.useEffect(() => {
   273→    setSlashSuppressed(false);
   274→    setSelIndex(0);
   275→    setAtSuppressed(false);
   276→    setAtIndex(0);
   277→  }, [draft]);
   278→
   279→  /* ---- slash command popup derivation ---- */
   280→  const slashWithIcons = React.useMemo<SlashCommandWithIcon[]>(
   281→    () => SLASH_COMMANDS as SlashCommandWithIcon[],
   282→    []
   283→  );
   284→  const isSlashDraft =
   285→    draft.startsWith("/") &&
   286→    !draft.slice(1).includes(" ") &&
   287→    !streaming;
   288→  const filtered = React.useMemo<SlashCommand[]>(() => {
   289→    if (!isSlashDraft) return [];
   290→    const q = draft.slice(1).toLowerCase();
   291→    if (!q) return slashWithIcons;
   292→    return slashWithIcons.filter(
   293→      (c) =>
   294→        c.label.toLowerCase().includes(q) ||
   295→        c.id.toLowerCase().includes(q) ||
   296→        c.hint.toLowerCase().includes(q)
   297→    );
   298→  }, [isSlashDraft, draft, slashWithIcons]);
   299→  const slashVisible = isSlashDraft && !slashSuppressed && filtered.length > 0;
   300→
   301→  /* ---- @ context reference popup derivation ---- */
   302→  const atQuery = React.useMemo<{ index: number; query: string } | null>(() => {
   303→    const lastAt = draft.lastIndexOf("@");
   304→    if (lastAt === -1) return null;
   305→    const after = draft.slice(lastAt + 1);
   306→    // the query is the text from the last "@" to the end of the draft;
   307→    // any whitespace terminates it.
   308→    if (/\s/.test(after)) return null;
   309→    return { index: lastAt, query: after };
   310→  }, [draft]);
   311→
   312→  const atFiltered = React.useMemo<RefItem[]>(() => {
   313→    if (!atQuery) return [];
   314→    const q = atQuery.query.toLowerCase();
   315→    const list = q
   316→      ? refItems.filter(
   317→          (it) =>
   318→            it.name.toLowerCase().includes(q) ||
   319→            it.path.toLowerCase().includes(q)
   320→        )
   321→      : refItems;
   322→    return list.slice(0, 12);
   323→  }, [atQuery, refItems]);
   324→
   325→  const atVisible =
   326→    !streaming &&
   327→    !slashVisible &&
   328→    atQuery !== null &&
   329→    !atSuppressed &&
   330→    atFiltered.length > 0;
   331→
   332→  // keep atIndex within bounds when the filtered list shrinks
   333→  React.useEffect(() => {
   334→    if (atFiltered.length > 0 && atIndex >= atFiltered.length) {
   335→      setAtIndex(0);
   336→    }
   337→  }, [atFiltered, atIndex]);
   338→
   339→  /* ---- actions ---- */
   340→  const handleSend = React.useCallback(() => {
   341→    if (streaming) return;
   342→    const text = useChatStore.getState().conversations[tabId]?.composerDraft ?? "";
   343→    if (!text.trim()) return;
   344→    void sendMessage(tabId, text, modelId);
   345→  }, [streaming, tabId, modelId, sendMessage]);
   346→
   347→  const selectCommand = React.useCallback(
   348→    (cmd: SlashCommand) => {
   349→      // Read the live draft + skills from the store to avoid stale closures.
   350→      const currentDraft =
   351→        useChatStore.getState().conversations[tabId]?.composerDraft ?? "";
   352→      const beforeSlash = currentDraft.slice(0, currentDraft.lastIndexOf("/"));
   353→      const focusTextarea = () =>
   354→        requestAnimationFrame(() => taRef.current?.focus());
   355→
   356→      if (cmd.id === "model") {
   357→        setComposerDraft(tabId, "");
   358→        setModelOpen(true);
   359→        return;
   360→      }
   361→      if (cmd.id === "clear") {
   362→        clearConversation(tabId);
   363→        setComposerDraft(tabId, "");
   364→        toast("Conversation cleared");
   365→        return;
   366→      }
   367→      if (cmd.id === "help") {
   368→        setComposerDraft(tabId, "");
   369→        toast("Press / for commands · Enter to send · Shift+Enter for newline", {
   370→          duration: 4000,
   371→        });
   372→        return;
   373→      }
   374→      if (cmd.id === "plan") {
   375→        setBuildMode("plan");
   376→        setComposerDraft(tabId, beforeSlash);
   377→        toast("Switched to Plan");
   378→        focusTextarea();
   379→        return;
   380→      }
   381→      if (cmd.id === "build") {
   382→        setBuildMode("build");
   383→        setComposerDraft(tabId, beforeSlash);
   384→        toast("Switched to Build");
   385→        focusTextarea();
   386→        return;
   387→      }
   388→      // action skills (review, test, fix, explain, refactor, document, init):
   389→      // add as a removable chip instead of inserting text.
   390→      const currentSkills =
   391→        useChatStore.getState().conversations[tabId]?.skills ?? [];
   392→      if (currentSkills.includes(cmd.id)) {
   393→        toast("Already added");
   394→      } else {
   395→        addSkill(tabId, cmd.id);
   396→        toast(`${stripSlash(cmd.label)} skill added`);
   397→      }
   398→      setComposerDraft(tabId, beforeSlash);
   399→      focusTextarea();
   400→    },
   401→    [
   402→      tabId,
   403→      setComposerDraft,
   404→      clearConversation,
   405→      addSkill,
   406→      setBuildMode,
   407→    ]
   408→  );
   409→
   410→  const selectRef = React.useCallback(
   411→    (item: RefItem) => {
   412→      if (!atQuery) return;
   413→      const before = draft.slice(0, atQuery.index);
   414→      const next = `${before}@${item.name} `;
   415→      setComposerDraft(tabId, next);
   416→      requestAnimationFrame(() => taRef.current?.focus());
   417→    },
   418→    [atQuery, draft, setComposerDraft, tabId]
   419→  );
   420→
   421→  /* ---- keyboard ---- */
   422→  const onKeyDown = (e: React.KeyboardEvent<HTMLTextAreaElement>) => {
   423→    if (slashVisible) {
   424→      if (e.key === "ArrowDown") {
   425→        e.preventDefault();
   426→        setSelIndex((i) => (i + 1) % filtered.length);
   427→        return;
   428→      }
   429→      if (e.key === "ArrowUp") {
   430→        e.preventDefault();
   431→        setSelIndex((i) => (i - 1 + filtered.length) % filtered.length);
   432→        return;
   433→      }
   434→      if (e.key === "Enter") {
   435→        e.preventDefault();
   436→        const cmd = filtered[selIndex] ?? filtered[0];
   437→        if (cmd) selectCommand(cmd);
   438→        return;
   439→      }
   440→      if (e.key === "Escape") {
   441→        e.preventDefault();
   442→        setSlashSuppressed(true);
   443→        return;
   444→      }
   445→      if (e.key === "Tab") {
   446→        e.preventDefault();
   447→        const cmd = filtered[selIndex] ?? filtered[0];
   448→        if (cmd) selectCommand(cmd);
   449→        return;
   450→      }
   451→    }
   452→    if (atVisible) {
   453→      if (e.key === "ArrowDown") {
   454→        e.preventDefault();
   455→        setAtIndex((i) => (i + 1) % atFiltered.length);
   456→        return;
   457→      }
   458→      if (e.key === "ArrowUp") {
   459→        e.preventDefault();
   460→        setAtIndex((i) => (i - 1 + atFiltered.length) % atFiltered.length);
   461→        return;
   462→      }
   463→      if (e.key === "Enter" || e.key === "Tab") {
   464→        e.preventDefault();
   465→        const item = atFiltered[atIndex] ?? atFiltered[0];
   466→        if (item) selectRef(item);
   467→        return;
   468→      }
   469→      if (e.key === "Escape") {
   470→        e.preventDefault();
   471→        setAtSuppressed(true);
   472→        return;
   473→      }
   474→    }
   475→    if (e.key === "Enter" && !e.shiftKey && !e.nativeEvent.isComposing) {
   476→      if (!streaming && draft.trim()) {
   477→        e.preventDefault();
   478→        handleSend();
   479→      }
   480→    }
   481→  };
   482→
   483→  /* ---- voice ---- */
   484→  const startVoice = React.useCallback(() => {
   485→    const w = window as WindowWithSpeech;
   486→    const Ctor = w.SpeechRecognition ?? w.webkitSpeechRecognition;
   487→    if (!Ctor) {
   488→      toast.error("Voice input not supported in this browser");
   489→      return;
   490→    }
   491→    try {
   492→      const rec = new Ctor();
   493→      rec.lang = "en-US";
   494→      rec.continuous = false;
   495→      rec.interimResults = false;
   496→      rec.onresult = (event: SpeechEventLike) => {
   497→        let transcript = "";
   498→        for (let i = 0; i < event.results.length; i++) {
   499→          transcript += event.results[i][0].transcript;
   500→        }
   501→        if (!transcript) return;
   502→        const cur =
   503→          useChatStore.getState().conversations[tabId]?.composerDraft ?? "";
   504→        const next = (cur ? cur + " " : "") + transcript.trim();
   505→        setComposerDraft(tabId, next);
   506→      };
   507→      rec.onerror = () => {
   508→        toast.error("Voice input failed");
   509→      };
   510→      rec.start();
   511→      toast("Listening…", { duration: 2500 });
   512→    } catch {
   513→      toast.error("Voice input could not start");
   514→    }
   515→  }, [tabId, setComposerDraft]);
   516→
   517→  /* ---- send button state ---- */
   518→  const canSend = draft.trim().length > 0;
   519→  const sendState: "send" | "stop" | "voice" = streaming
   520→    ? "stop"
   521→    : canSend
   522→    ? "send"
   523→    : "voice";
   524→
   525→  const onSendClick = () => {
   526→    if (sendState === "stop") stopStreaming(tabId);
   527→    else if (sendState === "send") handleSend();
   528→    else startVoice();
   529→  };
   530→
   531→  const SendIcon = sendState === "stop" ? Square : sendState === "send" ? ArrowUp : Mic;
   532→  const sendLabel = sendState === "stop" ? "Stop" : sendState === "send" ? "Send" : "Voice";
   533→
   534→  /* ---- active agent (for the trigger chip) ---- */
   535→  const activeAgent: AgentDef | undefined = agentId
   536→    ? AGENTS.find((a) => a.id === agentId)
   537→    : undefined;
   538→  const agentTriggerName = activeAgent ? activeAgent.name : "Agent";
   539→
   540→  const handleAgentSelect = (id: string, name: string) => {
   541→    setAgent(tabId, id);
   542→    setAgentOpen(false);
   543→    toast(`${name} selected`);
   544→  };
   545→
   546→  return (
   547→    <div className="relative w-full">
   548→      {/* slash command popup (above the box) */}
   549→      {slashVisible && (
   550→        <div
   551→          className={cn(
   552→            "absolute bottom-full mb-2 left-0 w-[300px] z-30",
   553→            "rounded-[12px] border border-ws-border bg-popover p-1.5 ws-fade-in",
   554→            "shadow-2xl max-h-[260px] overflow-y-auto ws-scroll"
   555→          )}
   556→          role="listbox"
   557→        >
   558→          {filtered.map((c, i) => {
   559→            const Icon = skillIcon(
   560→              (c as SlashCommandWithIcon).icon
   561→            );
   562→            return (
   563→              <button
   564→                type="button"
   565→                key={c.id}
   566→                onMouseEnter={() => setSelIndex(i)}
   567→                onClick={() => selectCommand(c)}
   568→                className={cn(
   569→                  "w-full flex items-center gap-2.5 rounded-[8px] px-2.5 py-2 cursor-pointer",
   570→                  "transition-colors text-left",
   571→                  i === selIndex ? "bg-hover" : "hover:bg-hover"
   572→                )}
   573→              >
   574→                <Icon
   575→                  className={cn(
   576→                    "size-4 shrink-0",
   577→                    i === selIndex ? "text-accent-hi" : "text-ws-secondary"
   578→                  )}
   579→                />
   580→                <div className="min-w-0">
   581→                  <div className="text-[13px] font-medium text-foreground">
   582→                    {c.label}
   583→                  </div>
   584→                  <div className="text-[12px] text-ws-secondary line-clamp-1">
   585→                    {c.description}
   586→                  </div>
   587→                </div>
   588→              </button>
   589→            );
   590→          })}
   591→        </div>
   592→      )}
   593→
   594→      {/* `@` context reference popup (above the box) */}
   595→      {atVisible && (
   596→        <div
