    82→function ProviderCard({
    83→  provider,
    84→  onRemove,
    85→}: {
    86→  provider: Provider;
    87→  onRemove: (p: Provider) => void;
    88→}) {
    89→  const updateProvider = useProvidersStore((s) => s.updateProvider);
    90→  const detectProviderModels = useProvidersStore((s) => s.detectProviderModels);
    91→  const isOllama = provider.type === "ollama";
    92→  const connected = isConnected(provider);
    93→  const [fetching, setFetching] = React.useState(false);
    94→
    95→  const onFetch = async () => {
    96→    if (!connected) return;
    97→    setFetching(true);
    98→    try {
    99→      const models = await detectProviderModels(provider.id);
   100→      if (models.length > 0) {
   101→        toast.success(
   102→          `${models.length} model(s) fetched from ${provider.name}`
   103→        );
   104→      } else {
   105→        toast.error("No models found — check the connection");
   106→      }
   107→    } catch {
   108→      toast.error("No models found — check the connection");
   109→    } finally {
   110→      setFetching(false);
   111→    }
   112→  };
   113→
   114→  return (
   115→    <div className="rounded-[10px] border border-ws-border bg-card-ws p-3">
   116→      <div className="flex items-center gap-2">
   117→        <Input
   118→          aria-label="Provider name"
   119→          value={provider.name}
   120→          onChange={(e) =>
   121→            updateProvider(provider.id, { name: e.target.value })
   122→          }
   123→          className="flex-1 h-8 bg-card-ws border-ws-border-soft text-[13px] focus-ring"
   124→        />
   125→        <span className="text-[11px] rounded-full px-1.5 py-0.5 border text-ws-secondary border-ws-border whitespace-nowrap">
   126→          {PROVIDER_TYPE_LABEL[provider.type]}
   127→        </span>
   128→        <Switch
   129→          checked={provider.enabled}
   130→          onCheckedChange={(v) =>
   131→            updateProvider(provider.id, { enabled: v })
   132→          }
   133→          aria-label="Toggle provider enabled"
   134→        />
   135→        <button
   136→          type="button"
   137→          onClick={() => onRemove(provider)}
   138→          aria-label={`Remove ${provider.name}`}
   139→          className="inline-flex items-center justify-center size-7 text-ws-secondary hover:text-red-400 hover:bg-hover rounded-[8px] focus-ring"
   140→        >
   141→          <Trash2 className="size-3.5" />
   142→        </button>
   143→      </div>
   144→
   145→      <div className="mt-2 grid gap-2">
   146→        <FieldRow label="Base URL" htmlFor={`prov-base-${provider.id}`}>
   147→          <Input
   148→            id={`prov-base-${provider.id}`}
   149→            value={provider.baseUrl}
   150→            onChange={(e) =>
   151→              updateProvider(provider.id, { baseUrl: e.target.value })
   152→            }
   153→            placeholder="https://api.openai.com/v1"
   154→            className="h-8 bg-card-ws border-ws-border-soft text-[13px] focus-ring"
   155→          />
   156→        </FieldRow>
   157→
   158→        {isOllama ? (
   159→          <p className="text-[12px] text-ws-secondary px-1 py-1">
   160→            Ollama runs locally and needs no API key. Use Fetch models to load
   161→            available models.
   162→          </p>
   163→        ) : (
   164→          <FieldRow label="API key" htmlFor={`prov-key-${provider.id}`}>
   165→            <Input
   166→              id={`prov-key-${provider.id}`}
   167→              type="password"
   168→              value={provider.apiKey}
   169→              onChange={(e) =>
   170→                updateProvider(provider.id, { apiKey: e.target.value })
   171→              }
   172→              onBlur={() => {
   173→                if (provider.apiKey.trim()) {
   174→                  void detectProviderModels(provider.id, true);
   175→                  toast(
   176→                    "Fetching models — all hidden until you select them"
   177→                  );
   178→                }
   179→              }}
   180→              placeholder="sk-..."
   181→              className="h-8 bg-card-ws border-ws-border-soft text-[13px] focus-ring"
   182→            />
   183→          </FieldRow>
   184→        )}
   185→      </div>
   186→
   187→      <div className="mt-2 flex items-center justify-between px-2">
   188→        <span className="text-[12px] text-ws-secondary">
   189→          {provider.models.length} model(s)
   190→        </span>
   191→        <button
   192→          type="button"
   193→          onClick={onFetch}
   194→          disabled={!connected || fetching}
   195→          className={cn(
   196→            "inline-flex items-center gap-2 h-7 px-2.5 rounded-[8px] bg-accent-fill text-white text-[12px] hover:opacity-90 focus-ring",
   197→            (!connected || fetching) && "opacity-50 cursor-not-allowed"
   198→          )}
   199→        >
   200→          {fetching ? (
   201→            <Loader2 className="size-3.5 animate-spin" />
   202→          ) : (
   203→            <RefreshCw className="size-3.5" />
   204→          )}
   205→          {fetching ? "Fetching…" : "Fetch models"}
   206→        </button>
   207→      </div>
   208→    </div>
   209→  );
   210→}

   212→function ModelsTabContent() {

   288→                      "inline-flex items-center gap-2 h-7 px-2.5 rounded-[8px] bg-accent-fill text-white text-[12px] hover:opacity-90 focus-ring",

   346→export function SettingsModal() {
   347→  const open = useAppStore((s) => s.settingsOpen);
   348→  const setOpen = useAppStore((s) => s.setSettingsOpen);
   349→
   350→  const apiKey = useSettingsStore((s) => s.apiKey);
   351→  const setApiKey = useSettingsStore((s) => s.setApiKey);
   352→  const baseUrl = useSettingsStore((s) => s.baseUrl);
   353→  const setBaseUrl = useSettingsStore((s) => s.setBaseUrl);
   354→  const defaultModelId = useSettingsStore((s) => s.defaultModelId);
   355→  const setDefaultModelId = useSettingsStore((s) => s.setDefaultModelId);
   356→  const theme = useSettingsStore((s) => s.theme);
   357→  const setTheme = useSettingsStore((s) => s.setTheme);
   358→  const agentInstructions = useSettingsStore((s) => s.agentInstructions);
   359→  const setAgentInstructions = useSettingsStore((s) => s.setAgentInstructions);
   360→  const skills = useSettingsStore((s) => s.skills);
   361→  const addSkill = useSettingsStore((s) => s.addSkill);
   362→  const removeSkill = useSettingsStore((s) => s.removeSkill);
   363→  const skillsFolderPath = useSettingsStore((s) => s.skillsFolderPath);
   364→  const setSkillsFolderPath = useSettingsStore((s) => s.setSkillsFolderPath);
   365→  const useRealApi = useSettingsStore((s) => s.useRealApi);
   366→  const setUseRealApi = useSettingsStore((s) => s.setUseRealApi);
   367→
   368→  const providers = useProvidersStore((s) => s.providers);
   369→  const addProvider = useProvidersStore((s) => s.addProvider);
   370→  const removeProvider = useProvidersStore((s) => s.removeProvider);
   371→
   372→  // Visible provider models (from connected providers, non-hidden) — used to
   373→  // populate the General tab's default-model picker instead of the old static
   374→  // built-in model list.
   375→  const visibleModels = React.useMemo(() => {
   376→    const out: { id: string; label: string }[] = [];
   377→    for (const p of providers) {
   378→      if (!isConnected(p)) continue;
   379→      for (const m of p.models) {
   380→        if (m.hidden) continue;
   381→        out.push({ id: m.id, label: m.name });
   382→      }
   383→    }
   384→    return out;
   385→  }, [providers]);
   386→
   387→  const [newSkill, setNewSkill] = React.useState("");
   388→  const [tab, setTab] = React.useState("general");
   389→
   390→  const [addOpen, setAddOpen] = React.useState(false);
   391→  const [newName, setNewName] = React.useState("");
   392→  const [newType, setNewType] = React.useState<ProviderType>("custom");
   393→  const [newBaseUrl, setNewBaseUrl] = React.useState("");
   394→  const [removeTarget, setRemoveTarget] = React.useState<Provider | null>(
   395→    null
   396→  );
   397→
   398→  const openAddProvider = () => {
   399→    setNewName("");
   400→    setNewType("custom");
   401→    setNewBaseUrl("");
   402→    setAddOpen(true);
   403→  };
   404→
   405→  const confirmAddProvider = () => {
   406→    const name = newName.trim();
   407→    if (!name) {
   408→      toast.error("Provider name is required");
   409→      return;
   410→    }
   411→    addProvider({ name, type: newType, baseUrl: newBaseUrl.trim() });
   412→    toast.success("Provider added");
   413→    setAddOpen(false);
   414→  };
   415→
   416→  const confirmRemove = () => {
   417→    if (!removeTarget) return;
   418→    removeProvider(removeTarget.id);
   419→    toast.success("Provider removed");
   420→    setRemoveTarget(null);
   421→  };
   422→
   423→  const addSkillOnEnter = (e: React.KeyboardEvent<HTMLInputElement>) => {
   424→    if (e.key !== "Enter") return;
   425→    e.preventDefault();
   426→    const v = newSkill.trim();
   427→    if (!v) return;
   428→    addSkill(v);
   429→    setNewSkill("");
   430→  };
   431→
   432→  const folderInputRef = React.useRef<HTMLInputElement>(null);
   433→
   434→  React.useEffect(() => {
   435→    const el = folderInputRef.current;
   436→    if (el) {
   437→      el.setAttribute("webkitdirectory", "");
   438→    }
   439→  }, []);
   440→
   441→  const onFolderPicked = (e: React.ChangeEvent<HTMLInputElement>) => {
   442→    const fileList = e.target.files;
   443→    if (!fileList || fileList.length === 0) {
   444→      e.target.value = "";
   445→      return;
   446→    }
   447→    const first = fileList[0] as File & { webkitRelativePath?: string };
   448→    const rel = first.webkitRelativePath;
   449→    if (rel) {
   450→      const folderName = rel.split("/")[0];
   451→      if (folderName) setSkillsFolderPath(folderName);
   452→    }
   453→    const added = new Set<string>();
   454→    for (let i = 0; i < fileList.length; i++) {
   455→      const name = fileList[i].name;
   456→      const lower = name.toLowerCase();
   457→      if (!lower.endsWith(".md") && !lower.endsWith(".txt")) continue;
   458→      const base = name.replace(/\.[^.]+$/, "");
   459→      if (!base) continue;
   460→      if (skills.includes(base) || added.has(base)) continue;
   461→      addSkill(base);
   462→      added.add(base);
   463→    }
   464→    if (added.size > 0) {
   465→      toast.success(`${added.size} skill(s) imported from folder`);
   466→    } else {
   467→      toast("No skill files found in folder");
   468→    }
   469→    e.target.value = "";
   470→  };
   471→
   472→  return (
   473→    <>
   474→    <Dialog open={open} onOpenChange={setOpen}>
   475→      <DialogContent className="max-w-[640px] p-0 bg-popover border border-ws-border">
   476→        <DialogHeader className="px-5 pt-5 pb-3 border-b border-ws-border">
   477→          <DialogTitle className="text-[15px]">Settings</DialogTitle>
   478→          <DialogDescription className="sr-only">
   479→            Configure API key, models, agent instructions and skills.
   480→          </DialogDescription>
   481→        </DialogHeader>
   482→
   483→        <div className="px-5 py-4">
   484→          <Tabs value={tab} onValueChange={setTab}>
   485→            <TabsList className="bg-card-ws border border-ws-border h-8">
   486→              <TabsTrigger value="general" className="text-[12.5px]">
   487→                General
   488→              </TabsTrigger>
   489→              <TabsTrigger value="providers" className="text-[12.5px]">
   490→                Providers
   491→              </TabsTrigger>
   492→              <TabsTrigger value="models" className="text-[12.5px]">
   493→                Models
   494→              </TabsTrigger>
   495→              <TabsTrigger value="agents" className="text-[12.5px]">
   496→                Agents
   497→              </TabsTrigger>
   498→              <TabsTrigger value="skills" className="text-[12.5px]">
   499→                Skills
   500→              </TabsTrigger>
   501→            </TabsList>
   502→
   503→            <TabsContent value="general" className="mt-4">
   504→              <div className="grid gap-4">
   505→                <FieldRow label="API key" htmlFor="settings-api-key">
   506→                  <Input
   507→                    id="settings-api-key"
   508→                    type="password"
   509→                    value={apiKey}
   510→                    onChange={(e) => setApiKey(e.target.value)}
   511→                    placeholder="sk-..."
   512→                    className="focus-ring bg-card-ws border-ws-border text-[13px]"
   513→                  />
   514→                </FieldRow>
   515→                <FieldRow label="Base URL" htmlFor="settings-base-url">
   516→                  <Input
   517→                    id="settings-base-url"
   518→                    value={baseUrl}
   519→                    onChange={(e) => setBaseUrl(e.target.value)}
   520→                    placeholder="https://api.openai.com/v1"
   521→                    className="focus-ring bg-card-ws border-ws-border text-[13px]"
   522→                  />
   523→                </FieldRow>
   524→                <div className="flex items-center justify-between rounded-lg border border-ws-border bg-card-ws px-3 py-2.5">
   525→                  <div className="grid gap-0.5">
   526→                    <span className="text-[13px] text-foreground">
   527→                      Use real OpenAI-compatible API
   528→                    </span>
   529→                    <span className="text-[12px] text-ws-disabled">
   530→                      When off, Goodstuff-Agent uses mock streaming responses.
   531→                    </span>
   532→                  </div>
   533→                  <Switch
   534→                    checked={useRealApi}
   535→                    onCheckedChange={setUseRealApi}
   536→                    aria-label="Use real API"
   537→                  />
   538→                </div>
   539→                <FieldRow label="Default model" htmlFor="settings-model">
   540→                  <Select
   541→                    value={
   542→                      visibleModels.some((m) => m.id === defaultModelId)
   543→                        ? defaultModelId
   544→                        : undefined
   545→                    }
   546→                    onValueChange={setDefaultModelId}
   547→                  >
   548→                    <SelectTrigger
   549→                      id="settings-model"
   550→                      className="focus-ring bg-card-ws border-ws-border text-[13px] w-full"
   551→                    >
   552→                      <SelectValue placeholder="Select default model" />
   553→                    </SelectTrigger>
   554→                    <SelectContent className="bg-popover border-ws-border">
   555→                      {visibleModels.length === 0 ? (
   556→                        <SelectItem
   557→                          value="__none__"
   558→                          disabled
   559→                          className="text-[13px]"
   560→                        >
   561→                          No models available
   562→                        </SelectItem>
   563→                      ) : (
   564→                        visibleModels.map((m) => (
   565→                          <SelectItem
   566→                            key={m.id}
   567→                            value={m.id}
   568→                            className="text-[13px]"
   569→                          >
   570→                            {m.label}
   571→                          </SelectItem>
   572→                        ))
   573→                      )}
   574→                    </SelectContent>
   575→                  </Select>
   576→                </FieldRow>
