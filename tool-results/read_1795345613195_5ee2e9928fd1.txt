    67→    67→/* ---- minimal Web Speech API typing (no `any`) ---- */
    68→    68→interface SpeechResultLike {
    69→    69→  0: { transcript: string };
    70→    70→  isFinal: boolean;
    71→    71→  length: number;
    72→    72→}
    73→    73→interface SpeechEventLike {
    74→    74→  results: ArrayLike<SpeechResultLike>;
    75→    75→}
    76→    76→interface SpeechRecognitionLike {
    77→    77→  lang: string;
    78→    78→  continuous: boolean;
    79→    79→  interimResults: boolean;
    80→    80→  start: () => void;
    81→    81→  stop: () => void;
    82→    82→  abort: () => void;
    83→    83→  onresult: ((event: SpeechEventLike) => void) | null;
    84→    84→  onerror: ((event: unknown) => void) | null;
    85→    85→  onend: (() => void) | null;
    86→    86→}
    87→    87→type SpeechRecognitionCtor = new () => SpeechRecognitionLike;
    88→    88→type WindowWithSpeech = Window & {
    89→    89→  SpeechRecognition?: SpeechRecognitionCtor;
    90→    90→  webkitSpeechRecognition?: SpeechRecognitionCtor;
    91→    91→};

   155→   155→/**
   156→   156→ * Composer — the core prompt input.
   157→   157→ * Auto-growing textarea, slash command popup (with skill icons) that adds
   158→   158→ * action skills as removable chips, Codex-style `@` context referencing,
   159→   159→ * attach menu, Plan/Build mode selector, agent picker chip, model picker
   160→   160→ * chip, and a send/stop/voice button.
   161→   161→ */
   162→   162→export function Composer({ tabId }: { tabId: string }) {
   163→   163→  const ensure = useChatStore((s) => s.ensure);
   164→   164→  const draft = useChatStore((s) => s.conversations[tabId]?.composerDraft ?? "");
   165→   165→  const streaming = useChatStore(
   166→   166→    (s) => s.conversations[tabId]?.streaming ?? false
   167→   167→  );
   168→   168→  const convModelId = useChatStore((s) => s.conversations[tabId]?.modelId);
   169→   169→  const defaultModelId = useSettingsStore((s) => s.defaultModelId);
   170→   170→  const setComposerDraft = useChatStore((s) => s.setComposerDraft);
   171→   171→  const sendMessage = useChatStore((s) => s.sendMessage);
   172→   172→  const stopStreaming = useChatStore((s) => s.stopStreaming);
   173→   173→  const clearConversation = useChatStore((s) => s.clearConversation);
   174→   174→
   175→   175→  /* ---- skills (multi-chip) ---- */
   176→   176→  const skills = useChatStore(
   177→   177→    (s) => s.conversations[tabId]?.skills ?? EMPTY_SKILLS
   178→   178→  );
   179→   179→  const addSkill = useChatStore((s) => s.addSkill);
   180→   180→  const removeSkill = useChatStore((s) => s.removeSkill);
   181→   181→
   182→   182→  /* ---- agent picker ---- */
   183→   183→  const agentId = useChatStore(
   184→   184→    (s) => s.conversations[tabId]?.agentId ?? ""
   185→   185→  );
   186→   186→  const setAgent = useChatStore((s) => s.setAgent);
   187→   187→
   188→   188→  /* ---- Plan / Build mode (defensive: works with or without store support) ---- */
   189→   189→  const storeBuildMode = useChatStore(
   190→   190→    (s) =>
   191→   191→      (s.conversations[tabId] as ConvWithBuild | undefined)?.buildMode ??
   192→   192→      ("build" as BuildMode)
   193→   193→  );
   194→   194→  const storeSetBuildMode = useChatStore(
   195→   195→    (s) => (s as unknown as ChatStoreWithBuild).setBuildMode
   196→   196→  );
   197→   197→  const [localBuildMode, setLocalBuildMode] =
   198→   198→    React.useState<BuildMode>("build");
   199→   199→  const buildMode: BuildMode = storeSetBuildMode
   200→   200→    ? (storeBuildMode as BuildMode)
   201→   201→    : localBuildMode;
   202→   202→  const setBuildMode = React.useCallback(
   203→   203→    (mode: BuildMode) => {
   204→   204→      if (storeSetBuildMode) storeSetBuildMode(tabId, mode);
   205→   205→      else setLocalBuildMode(mode);
   206→   206→    },
   207→   207→    [storeSetBuildMode, tabId]
   208→   208→  );
   209→   209→
   210→   210→  /* ---- active workspace (reactive) for `@` referencing ---- */
   211→   211→  const projectId = useAppStore(
   212→   212→    (s) => s.tabs.find((t) => t.id === tabId)?.projectId
   213→   213→  );
   214→   214→  const projects = useProjectsStore((s) => s.projects);
   215→   215→  const setTabProject = useAppStore((s) => s.setTabProject);
   216→   216→  const addProject = useProjectsStore((s) => s.addProject);
   217→   217→  const activeProject = projects.find((p) => p.id === projectId);
   218→   218→  const convFiles = useChatStore(
   219→   219→    (s) => s.conversations[tabId]?.workspace.files
   220→   220→  );
   221→   221→
   222→   222→  const files = React.useMemo<FileNode[]>(() => {
   223→   223→    if (projectId) {
   224→   224→      return (
   225→   225→        projects.find((p) => p.id === projectId)?.workspace.files ??
   226→   226→        EMPTY_FILES
   227→   227→      );
   228→   228→    }
   229→   229→    return convFiles ?? EMPTY_FILES;
   230→   230→  }, [projectId, projects, convFiles]);
   231→   231→
   232→   232→  const refItems = React.useMemo<RefItem[]>(() => {
   233→   233→    const fileItems: RefItem[] = files.map((f) => ({
   234→   234→      kind: "file",
   235→   235→      name: f.name,
   236→   236→      path: f.path,
   237→   237→      icon: File,
   238→   238→    }));
   239→   239→    const projItems: RefItem[] = projects.map((p) => ({
   240→   240→      kind: "project",
   241→   241→      name: p.name,
   242→   242→      path: p.name,
   243→   243→      icon: Folder,
   244→   244→    }));
   245→   245→    return [...fileItems, ...projItems];
   246→   246→  }, [files, projects]);
   247→   247→
   248→   248→  const modelId = convModelId ?? defaultModelId;
   249→   249→
   250→   250→  const taRef = React.useRef<HTMLTextAreaElement>(null);
   251→   251→  const [modelOpen, setModelOpen] = React.useState(false);
   252→   252→  const [agentOpen, setAgentOpen] = React.useState(false);
   253→   253→  const [slashSuppressed, setSlashSuppressed] = React.useState(false);
   254→   254→  const [selIndex, setSelIndex] = React.useState(0);
   255→   255→  const [atSuppressed, setAtSuppressed] = React.useState(false);
   256→   256→  const [atIndex, setAtIndex] = React.useState(0);
   257→   257→
   258→   258→  // ensure a conversation object exists so modelId/draft are stable
   259→   259→  React.useEffect(() => {
   260→   260→    ensure(tabId);
   261→   261→  }, [tabId, ensure]);
   262→   262→
   263→   263→  // auto-grow textarea whenever draft changes (covers typing + external updates)
   264→   264→  React.useLayoutEffect(() => {
   265→   265→    const ta = taRef.current;
   266→   266→    if (!ta) return;
   267→   267→    ta.style.height = "auto";
   268→   268→    ta.style.height = `${Math.min(ta.scrollHeight, 200)}px`;
   269→   269→  }, [draft]);
   270→   270→
   271→   271→  // reset slash + @ suppression + selection whenever draft changes
   272→   272→  React.useEffect(() => {
   273→   273→    setSlashSuppressed(false);
   274→   274→    setSelIndex(0);
   275→   275→    setAtSuppressed(false);
   276→   276→    setAtIndex(0);
   277→   277→  }, [draft]);
   278→   278→
   279→   279→  /* ---- slash command popup derivation ---- */
   280→   280→  const slashWithIcons = React.useMemo<SlashCommandWithIcon[]>(
   281→   281→    () => SLASH_COMMANDS as SlashCommandWithIcon[],
   282→   282→    []
   283→   283→  );
   284→   284→  const isSlashDraft =
   285→   285→    draft.startsWith("/") &&
   286→   286→    !draft.slice(1).includes(" ") &&
   287→   287→    !streaming;
   288→   288→  const filtered = React.useMemo<SlashCommand[]>(() => {
   289→   289→    if (!isSlashDraft) return [];
   290→   290→    const q = draft.slice(1).toLowerCase();
   291→   291→    if (!q) return slashWithIcons;
   292→   292→    return slashWithIcons.filter(
   293→   293→      (c) =>
   294→   294→        c.label.toLowerCase().includes(q) ||
   295→   295→        c.id.toLowerCase().includes(q) ||
   296→   296→        c.hint.toLowerCase().includes(q)
   297→   297→    );
   298→   298→  }, [isSlashDraft, draft, slashWithIcons]);
   299→   299→  const slashVisible = isSlashDraft && !slashSuppressed && filtered.length > 0;
   300→   300→
   301→   301→  /* ---- @ context reference popup derivation ---- */
   302→   302→  const atQuery = React.useMemo<{ index: number; query: string } | null>(() => {
   303→   303→    const lastAt = draft.lastIndexOf("@");
   304→   304→    if (lastAt === -1) return null;
   305→   305→    const after = draft.slice(lastAt + 1);
   306→   306→    // the query is the text from the last "@" to the end of the draft;
   307→   307→    // any whitespace terminates it.
   308→   308→    if (/\s/.test(after)) return null;
   309→   309→    return { index: lastAt, query: after };
   310→   310→  }, [draft]);
   311→   311→
   312→   312→  const atFiltered = React.useMemo<RefItem[]>(() => {
   313→   313→    if (!atQuery) return [];
   314→   314→    const q = atQuery.query.toLowerCase();
   315→   315→    const list = q
   316→   316→      ? refItems.filter(
   317→   317→          (it) =>
   318→   318→            it.name.toLowerCase().includes(q) ||
   319→   319→            it.path.toLowerCase().includes(q)
   320→   320→        )
   321→   321→      : refItems;
   322→   322→    return list.slice(0, 12);
   323→   323→  }, [atQuery, refItems]);
   324→   324→
   325→   325→  const atVisible =
   326→   326→    !streaming &&
   327→   327→    !slashVisible &&
   328→   328→    atQuery !== null &&
   329→   329→    !atSuppressed &&
   330→   330→    atFiltered.length > 0;
   331→   331→
   332→   332→  // keep atIndex within bounds when the filtered list shrinks
   333→   333→  React.useEffect(() => {
   334→   334→    if (atFiltered.length > 0 && atIndex >= atFiltered.length) {
   335→   335→      setAtIndex(0);
   336→   336→    }
   337→   337→  }, [atFiltered, atIndex]);
   338→   338→
   339→   339→  /* ---- actions ---- */
   340→   340→  const handleSend = React.useCallback(() => {
   341→   341→    if (streaming) return;
   342→   342→    const text = useChatStore.getState().conversations[tabId]?.composerDraft ?? "";
   343→   343→    if (!text.trim()) return;
   344→   344→    void sendMessage(tabId, text, modelId);
   345→   345→  }, [streaming, tabId, modelId, sendMessage]);
   346→   346→
   347→   347→  const selectCommand = React.useCallback(
   348→   348→    (cmd: SlashCommand) => {
   349→   349→      // Read the live draft + skills from the store to avoid stale closures.
   350→   350→      const currentDraft =
   351→   351→        useChatStore.getState().conversations[tabId]?.composerDraft ?? "";
   352→   352→      const beforeSlash = currentDraft.slice(0, currentDraft.lastIndexOf("/"));
   353→   353→      const focusTextarea = () =>
   354→   354→        requestAnimationFrame(() => taRef.current?.focus());
   355→   355→
   356→   356→      if (cmd.id === "model") {
   357→   357→        setComposerDraft(tabId, "");
   358→   358→        setModelOpen(true);
   359→   359→        return;
   360→   360→      }
   361→   361→      if (cmd.id === "clear") {
   362→   362→        clearConversation(tabId);
   363→   363→        setComposerDraft(tabId, "");
   364→   364→        toast("Conversation cleared");
   365→   365→        return;
   366→   366→      }
   367→   367→      if (cmd.id === "help") {
   368→   368→        setComposerDraft(tabId, "");
   369→   369→        toast("Press / for commands · Enter to send · Shift+Enter for newline", {
   370→   370→          duration: 4000,
   371→   371→        });
   372→   372→        return;
   373→   373→      }
   374→   374→      if (cmd.id === "plan") {
   375→   375→        setBuildMode("plan");
   376→   376→        setComposerDraft(tabId, beforeSlash);
   377→   377→        toast("Switched to Plan");
   378→   378→        focusTextarea();
   379→   379→        return;
   380→   380→      }
   381→   381→      if (cmd.id === "build") {
   382→   382→        setBuildMode("build");
   383→   383→        setComposerDraft(tabId, beforeSlash);
   384→   384→        toast("Switched to Build");
   385→   385→        focusTextarea();
   386→   386→        return;
   387→   387→      }
   388→   388→      // action skills (review, test, fix, explain, refactor, document, init):
   389→   389→      // add as a removable chip instead of inserting text.
   390→   390→      const currentSkills =
   391→   391→        useChatStore.getState().conversations[tabId]?.skills ?? [];
   392→   392→      if (currentSkills.includes(cmd.id)) {
   393→   393→        toast("Already added");
   394→   394→      } else {
   395→   395→        addSkill(tabId, cmd.id);
   396→   396→        toast(`${stripSlash(cmd.label)} skill added`);
   397→   397→      }
   398→   398→      setComposerDraft(tabId, beforeSlash);
   399→   399→      focusTextarea();
   400→   400→    },
   401→   401→    [
   402→   402→      tabId,
   403→   403→      setComposerDraft,
   404→   404→      clearConversation,
   405→   405→      addSkill,
   406→   406→      setBuildMode,
   407→   407→    ]
   408→   408→  );
   409→   409→
   410→   410→  const selectRef = React.useCallback(
   411→   411→    (item: RefItem) => {
   412→   412→      if (!atQuery) return;
   413→   413→      const before = draft.slice(0, atQuery.index);
   414→   414→      const next = `${before}@${item.name} `;
   415→   415→      setComposerDraft(tabId, next);
   416→   416→      requestAnimationFrame(() => taRef.current?.focus());
   417→   417→    },
   418→   418→    [atQuery, draft, setComposerDraft, tabId]
   419→   419→  );
   420→   420→
   421→   421→  /* ---- keyboard ---- */
   422→   422→  const onKeyDown = (e: React.KeyboardEvent<HTMLTextAreaElement>) => {
   423→   423→    if (slashVisible) {
   424→   424→      if (e.key === "ArrowDown") {
   425→   425→        e.preventDefault();
   426→   426→        setSelIndex((i) => (i + 1) % filtered.length);
   427→   427→        return;
   428→   428→      }
   429→   429→      if (e.key === "ArrowUp") {
   430→   430→        e.preventDefault();
   431→   431→        setSelIndex((i) => (i - 1 + filtered.length) % filtered.length);
   432→   432→        return;
   433→   433→      }
   434→   434→      if (e.key === "Enter") {
   435→   435→        e.preventDefault();
   436→   436→        const cmd = filtered[selIndex] ?? filtered[0];
   437→   437→        if (cmd) selectCommand(cmd);
   438→   438→        return;
   439→   439→      }
   440→   440→      if (e.key === "Escape") {
   441→   441→        e.preventDefault();
   442→   442→        setSlashSuppressed(true);
   443→   443→        return;
   444→   444→      }
   445→   445→      if (e.key === "Tab") {
   446→   446→        e.preventDefault();
   447→   447→        const cmd = filtered[selIndex] ?? filtered[0];
   448→   448→        if (cmd) selectCommand(cmd);
   449→   449→        return;
   450→   450→      }
   451→   451→    }
   452→   452→    if (atVisible) {
   453→   453→      if (e.key === "ArrowDown") {
   454→   454→        e.preventDefault();
   455→   455→        setAtIndex((i) => (i + 1) % atFiltered.length);
   456→   456→        return;
   457→   457→      }
   458→   458→      if (e.key === "ArrowUp") {
   459→   459→        e.preventDefault();
   460→   460→        setAtIndex((i) => (i - 1 + atFiltered.length) % atFiltered.length);
   461→   461→        return;
   462→   462→      }
   463→   463→      if (e.key === "Enter" || e.key === "Tab") {
   464→   464→        e.preventDefault();
   465→   465→        const item = atFiltered[atIndex] ?? atFiltered[0];
   466→   466→        if (item) selectRef(item);
   467→   467→        return;
   468→   468→      }
   469→   469→      if (e.key === "Escape") {
   470→   470→        e.preventDefault();
   471→   471→        setAtSuppressed(true);
   472→   472→        return;
   473→   473→      }
   474→   474→    }
   475→   475→    if (e.key === "Enter" && !e.shiftKey && !e.nativeEvent.isComposing) {
   476→   476→      if (!streaming && draft.trim()) {
   477→   477→        e.preventDefault();
   478→   478→        handleSend();
   479→   479→      }
   480→   480→    }
   481→   481→  };
   482→   482→
   483→   483→  /* ---- voice ---- */
   484→   484→  const startVoice = React.useCallback(() => {
   485→   485→    const w = window as WindowWithSpeech;
   486→   486→    const Ctor = w.SpeechRecognition ?? w.webkitSpeechRecognition;
   487→   487→    if (!Ctor) {
   488→   488→      toast.error("Voice input not supported in this browser");
   489→   489→      return;
   490→   490→    }
